   119→    const q = filterText.trim().toLowerCase();

   121→      if (r.kind === 'excluded') return false;
   122→      if (activeTab === 'unmapped' && r.kind !== 'unmapped') return false;
   123→      if (activeTab === 'income' && r.category !== 'income') return false;
   124→      if (activeTab === 'deductions' && r.category !== 'deductions') return false;
   125→      if (activeTab === 'assets' && r.category !== 'assets') return false;
   126→      if (filterMapped === 'mapped' && r.kind !== 'mapped') return false;
   127→      if (filterMapped === 'unmapped' && r.kind !== 'unmapped') return false;
   128→      if (q && !r.name.toLowerCase().includes(q)) return false;

   134→    const map = new Map<string, TaxRow[]>();
   135→    for (const r of filteredRows) {
   136→      if (!map.has(r.section)) map.set(r.section, []);
   137→      map.get(r.section)!.push(r);

   202→    if (!mapLine) {
   203→      onToast('Select a tax line');

   211→
   212→    if (andNext) {
   213→      const next = rows.find(r => r.kind === 'unmapped' && r.id !== savedRowId);
   214→      if (next) {
   215→        setCurrentRowId(next.id);
   216→        setMapLine('');
   217→        return;
   218→      }
   219→    }
   220→    setMapOpen(false);
   221→    setCurrentRowId(null);
   222→    setMapLine('');

   241→  function handleRowMenuAction(action: string) {
   242→    setRowMenuOpen(false);
   243→    if (!currentMenuRow) return;

   245→    switch (action) {
   246→      case 'remap':
   247→        openMapDrawer(row);
   248→        break;
   249→      case 'view-register':
   250→        onToast(`Register → ${row.name}`);
   251→        break;
   252→      case 'run-report':
   253→        onToast(`Run report for ${row.name}`);
   254→        break;
   255→      case 'view-transactions':
   256→        onToast(`Viewing transactions for ${row.name}`);
   257→        break;
   258→      case 'export-line':
   259→        onToast(`Exporting line: ${row.name}`);
   260→        break;
   261→      case 'flag-review': {
   262→        const name = row.name;
   263→        setRows(prev => prev.map(r => r.id === row.id ? { ...r, flagged: true } : r));
   264→        onToast(`${name} flagged for accountant review`);
   265→        break;
   266→      }
   267→      case 'exclude': {
   268→        const name = row.name;
   269→        setRows(prev => prev.map(r => r.id === row.id ? { ...r, kind: 'excluded' } : r));
   270→        onToast(`${name} excluded from tax summary`);
   271→        break;
   272→      }
   273→    }
   274→  }

   615→/* ─────────────────────────────────────────────────────────────
   616→   Section block sub-component (keeps the table render readable)
   617→   ───────────────────────────────────────────────────────────── */
   618→
   619→interface SectionBlockProps {
   620→  section: SectionDef;
   621→  rows: TaxRow[];
   622→  onRemap: (row: TaxRow) => void;
   623→  onRowClick: (row: TaxRow, e: React.MouseEvent<HTMLTableRowElement>) => void;
   624→  onRowMenuBtnClick: (row: TaxRow, e: React.MouseEvent<HTMLButtonElement>) => void;
   625→}
   626→
   627→function SectionBlock({ section, rows, onRemap, onRowClick, onRowMenuBtnClick }: SectionBlockProps) {
   628→  return (
   629→    <>
   630→      <tr className="section"><td colSpan={5}>{section.label}</td></tr>
   631→      {rows.map(row => {
   632→        const isUnmapped = row.kind === 'unmapped';
   633→        const rowCls = [
   634→          'clickable',
   635→          isUnmapped ? 'unmapped' : '',
   636→          row.flagged ? 'flagged' : '',
   637→        ].filter(Boolean).join(' ');
   638→        return (
   639→          <tr key={row.id} className={rowCls} onClick={e => onRowClick(row, e)}>
   640→            <td>
   641→              {row.name}
   642→              <div className="text-[11px] text-on-surface-variant">{row.formLine}</div>
   643→            </td>
   644→            <td className="right">{row.amount}</td>
   645→            <td className="right">{row.ytd}</td>
   646→            <td>
   647→              <span className={`mapping-badge ${row.kind === 'unmapped' ? 'unmapped' : 'mapped'}`}>
   648→                <span className="material-symbols-outlined text-[11px]">
   649→                  {row.kind === 'unmapped' ? 'error' : 'check_circle'}
   650→                </span>
   651→                {row.mapping}
   652→              </span>
   653→            </td>
   654→            <td className="text-right">
   655→              <button
   656→                className="text-[11px] font-semibold text-primary hover:underline"
   657→                onClick={e => { e.stopPropagation(); onRemap(row); }}
   658→              >
   659→                {isUnmapped ? 'Map now' : 'Remap'}
   660→              </button>
   661→              <button
   662→                data-row-menu-btn
   663→                className="ml-2 inline-flex items-center justify-center w-5 h-5 rounded hover:bg-surface-container"
   664→                onClick={e => onRowMenuBtnClick(row, e)}
   665→              >
   666→                <span className="material-symbols-outlined text-[14px] text-on-surface-variant">more_horiz</span>
   667→              </button>
   668→            </td>
   669→          </tr>
   670→        );
   671→      })}
   672→      <tr className="total">
   673→        <td>{section.totalLabel}</td>
   674→        <td className="right">{section.totalAmount}</td>
   675→        <td className="right">{section.totalYtd}</td>
   676→        <td colSpan={2}></td>
   677→      </tr>
   678→    </>
   679→  );
   680→}
