   818→function applyFilters(){
   819→  const q = document.querySelector("[data-filter]").value.toLowerCase();
   820→  let shown = 0, total = 0;
   821→  document.querySelectorAll(".v-row").forEach(tr=>{
   822→    total++;
   823→    const nameOk = !q || (tr.dataset.name||"").toLowerCase().includes(q);
   824→    let tabOk = true;
   825→    if (currentTab === "Suppliers") tabOk = tr.dataset.role === "Supplier";
   826→    if (currentTab === "Contractors") tabOk = tr.dataset.role === "Contractor";
   827→    if (currentTab === "1099") tabOk = tr.dataset["1099"] === "true";
   828→    if (currentTab === "missing-w9") tabOk = tr.dataset.w9 === "off" || tr.dataset.w9 === "pending" || tr.dataset.w9 === "missing";
   829→    const vis = nameOk && tabOk;
   830→    tr.style.display = vis ? "" : "none";
   831→    if (vis) shown++;
   832→  });
   833→  document.querySelector("[data-v-count]").textContent = `1–${shown} of ${total}`;
   834→}
