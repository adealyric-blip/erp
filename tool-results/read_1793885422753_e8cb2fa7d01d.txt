   113→    () => () => {
   114→      if (toastTimer.current !== null) window.clearTimeout(toastTimer.current);
   115→    },

   118→

   121→  const kpiPaused = rows.filter(r => r.status === 'paused').length;

   138→      setRuleName(rule.dataName);
   139→      setCondRows(cs => cs.map(c => (c.kind === 'first' ? { ...c, value: rule.keyword } : c)));

   142→      setEditingRow(null);
   143→      setRuleName('');
   144→      setCondRows(cs => cs.map(c => (c.kind === 'first' ? { ...c, value: '' } : c)));

   158→      { id: nextCondId.current++, kind: 'added', field: 'Bank description', op: 'Contains', value: '' },

   185→      id: nextRowId.current++,
   186→      name,
   187→      dataName: name,
   188→      keyword: kw.toUpperCase(),
   189→      category: cat,
   190→      catClass: 'bg-qb-blue/10 text-qb-blue',
   191→      matched: 0,
   192→      status: 'active',
   193→    };
   194→    const next = [tr, ...rows];
   195→    setRows(next);
   196→    updateKpis(next);

   199→      setDrawerOpen(false);

   201→      setRuleName('');
   202→      setCondRows(cs => cs.map(c => (c.kind === 'first' ? { ...c, value: '' } : c)));
   203→      setThenCategory('');
   204→    }

   211→  };

   214→    const isOn = row.status !== 'active'; // status after the flip

   216→      r.id === row.id ? { ...r, status: (isOn ? 'active' : 'paused') as RuleStatus } : r

   218→    setRows(next);
   219→    updateKpis(next);
   220→    toast(isOn ? 'Rule activated' : 'Rule paused');

   226→      rowId: row.id,
   227→      top: r.bottom + 4,
   228→      left: Math.min(r.right - 260, window.innerWidth - 280),
   229→      paused: row.status === 'paused',
   230→    });
   231→  };
   232→
   233→  const handleMenuAction = (action: string, tr: RuleRow) => {
   234→    setMenu(null);
   235→    switch (action) {
   236→      case 'edit':
   237→      case 'duplicate':
   238→        openDrawer(tr);
   239→        break;
   240→      case 'apply-existing': {
   241→        const matched = Math.floor(Math.random() * 8) + 3;
   242→        toast(`Applied to ${matched} past transactions`);
   243→        setRows(rows.map(r => (r.id === tr.id ? { ...r, matched: r.matched + matched } : r)));
   244→        break;
   245→      }
   246→      case 'view-matched':
   247→        toast(`Showing transactions matched by "${tr.dataName}"`);
   248→        break;
   249→      case 'pause': {
   250→        const next = rows.map(r => (r.id === tr.id ? { ...r, status: 'paused' as RuleStatus } : r));
   251→        setRows(next);
   252→        updateKpis(next);
   253→        toast('Rule paused');
   254→        break;
   255→      }
   256→      case 'activate': {
   257→        const next = rows.map(r => (r.id === tr.id ? { ...r, status: 'active' as RuleStatus } : r));
   258→        setRows(next);
   259→        updateKpis(next);
   260→        toast('Rule activated');
   261→        break;
   262→      }
   263→      case 'delete': {
   264→        const next = rows.filter(r => r.id !== tr.id);
   265→        setRows(next);
   266→        updateKpis(next);
   267→        toast('Rule deleted');
   268→        break;
   269→      }
   270→    }
   271→  };
   272→
   273→  /* ── View tabs ── */
   274→  const handleViewTab = (st: 'all' | 'active' | 'paused') => {
   275→    setViewTab(st);
   276→    const shown = st === 'all' ? rows.length : rows.filter(r => r.status === st).length;
   277→    setRlCount(`1–${shown} of ${rows.length}`);
   278→  };
   279→  const tabClassName = (key: 'all' | 'active' | 'paused', variant: boolean) =>
   280→    `h-full px-3 text-label-md tab-underline${variant ? ' text-on-surface-variant' : ''}${
   281→      viewTab === key ? ' active' : ''
   282→    }`;
   283→
   284→  /* ── Bulk bar ── */
   285→  const toggleSelected = (id: number) => {
   286→    setSelectedIds(prev => {
   287→      const next = new Set(prev);
   288→      if (next.has(id)) next.delete(id);
   289→      else next.add(id);
   290→      return next;
   291→    });
   292→  };
   293→
   294→  const handleCheckAll = (checked: boolean) => {
   295→    setCheckAll(checked);
   296→    setSelectedIds(checked ? new Set(rows.map(r => r.id)) : new Set());
   297→  };
   298→
   299→  const handleBulk = (act: string) => {
   300→    if (act === 'close') {
   301→      setSelectedIds(new Set());
   302→      setCheckAll(false);
   303→      return;
   304→    }
   305→    const n = selectedIds.size;
   306→    if (act === 'delete') {
   307→      const next = rows.filter(r => !selectedIds.has(r.id));
   308→      setRows(next);
   309→      updateKpis(next);
   310→    }
   311→    if (act === 'activate') {
   312→      const next = rows.map(r =>
   313→        selectedIds.has(r.id) ? { ...r, status: 'active' as RuleStatus } : r
   314→      );
   315→      setRows(next);
   316→      updateKpis(next);
   317→    }
   318→    if (act === 'pause') {
   319→      const next = rows.map(r =>
   320→        selectedIds.has(r.id) ? { ...r, status: 'paused' as RuleStatus } : r
   321→      );
   322→      setRows(next);
   323→      updateKpis(next);
   324→    }
   325→    toast(`${act.charAt(0).toUpperCase() + act.slice(1)} on ${n} rule${n !== 1 ? 's' : ''}`);

   327→    setCheckAll(false);

   347→          <span className="material-symbols-outlined text-[20px] text-on-surface-variant">account_balance</span>
   348→          <span className="text-label-md font-semibold">Finance</span>
   349→          <span className="text-outline">/</span>
   350→          <span className="text-label-md">Accounting</span>
   351→          <span className="text-outline">/</span>
   352→          <span className="text-label-md text-primary font-semibold">Rules</span>

   356→      <main className="px-6 pb-12">

   358→        {/* Page header */}

   363→          </div>
   364→          <div className="flex items-center gap-2">
   365→            <button className="h-8 px-3 border border-surface-dim rounded text-body-md hover:bg-surface-container-low flex items-center gap-1" data-export>
   366→              <span className="material-symbols-outlined text-[16px]">download</span>Export</button>

   372→        {/* KPI strip */}
   373→        <div className="grid grid-cols-2 xl:grid-cols-4 gap-3 mb-5">

   392→        {/* Table card */}

   395→          <div className="flex items-center gap-1 px-4 h-11 border-b border-surface-dim overflow-x-auto">

   437→                <tr key={row.id} className={`hover:bg-surface-container-low cursor-pointer rl-row${row.status === 'paused' ? ' opacity-70' : ''}`} data-status={row.status} data-name={row.dataName} data-category={row.category} onClick={e => handleRowClick(e, row)}>

   441→                  <td className="px-3 py-2"><span className={`px-2 py-0.5 rounded-full text-[11px] ${row.catClass} font-medium`}>{row.category}</span></td>

   443→                  <td className="px-3 py-2"><div className={`switch${row.status === 'active' ? ' on' : ''}`} data-toggle onClick={e => { e.stopPropagation(); toggleRule(row); }} /></td>
   444→                  <td className="px-2 py-2 text-right"><button className="w-6 h-6 rounded hover:bg-surface-container flex items-center justify-center" data-row-menu aria-label="Rule actions" onClick={e => { e.stopPropagation(); openRowMenu(e, row); }}><span className="material-symbols-outlined text-[16px] text-on-surface-variant">more_horiz</span></button></td>
   445→                </tr>
   446→              ))}
   447→            </tbody>
   448→          </table>
   449→        </div>
   450→
   451→      </main>
   452→
   453→      {/* ═══════════ RULE EDITOR DRAWER ═══════════ */}
   454→      <div className={drawerOpen ? 'drawer-scrim open' : 'drawer-scrim'} id="rule-scrim" onClick={closeDrawer} />
   455→      <div className={drawerOpen ? 'drawer open' : 'drawer'} id="rule-drawer">
   456→        <div className="flex items-center justify-between px-5 h-14 border-b border-surface-dim shrink-0">
   457→          <div className="flex items-center gap-2 min-w-0">
   458→            <span className="material-symbols-outlined text-[20px] text-primary">tune</span>
   459→            <div className="min-w-0">
   460→              <div className="text-headline-sm truncate" data-rule-title>{editingRow ? 'Edit rule' : 'New bank rule'}</div>
   461→              <div className="text-[11px] text-on-surface-variant">When the bank description matches, categorize</div>
   462→            </div>
   463→          </div>
   464→          <button className="w-8 h-8 rounded hover:bg-surface-container flex items-center justify-center" data-close-rule aria-label="Close" onClick={closeDrawer}>
   465→            <span className="material-symbols-outlined text-[18px]">close</span></button>
   466→        </div>
   467→
   468→        <div className="flex-1 overflow-y-auto p-5 space-y-5">
   469→
   470→          {/* Rule name */}
   471→          <div>
   472→            <label className="field-label">Rule name</label>
   473→            <input className="field-input" value={ruleName} onChange={e => setRuleName(e.target.value)} placeholder="e.g. Staples → Office supplies" data-rule-name />
   474→          </div>

   478→            <div className="flex items-center justify-between mb-2">
   479→              <div className="text-label-sm text-on-surface-variant uppercase tracking-wider font-bold">When (conditions)</div>

   482→                <option>Match ANY condition</option>
   483→              </select>
   484→            </div>
   485→            <div data-conditions>
   486→              {condRows.map(c => (
   487→                <div className="cond-row" key={c.id}>
   488→                  <span className="cond-prefix">{c.kind === 'first' ? 'When' : 'And'}</span>
   489→                  <select className="cond-select" value={c.field} onChange={e => updateCond(c.id, { field: e.target.value })}>

   491→                  </select>
   492→                  <select className="cond-select" value={c.op} onChange={e => updateCond(c.id, { op: e.target.value })}>
   493→                    {(c.kind === 'first' ? FIRST_OP_OPTIONS : ADDED_OP_OPTIONS).map(o => <option key={o}>{o}</option>)}
   494→                  </select>
   495→                  <input className="cond-input" placeholder={c.kind === 'first' ? 'e.g. STAPLES' : 'e.g. $0'} data-cond-val={c.kind === 'first' ? '' : undefined} value={c.value} onChange={e => updateCond(c.id, { value: e.target.value })} />
   496→                  <button className="cond-remove" data-remove-cond aria-label="Remove condition" onClick={() => removeCond(c.id)}>
   497→                    <span className="material-symbols-outlined text-[16px]">close</span></button>
   498→                </div>
   499→              ))}
   500→            </div>
   501→            <button className="h-7 px-2 mt-2 border border-dashed border-surface-dim rounded text-[11px] font-semibold text-primary hover:bg-surface-container-low flex items-center gap-1" data-add-cond onClick={addCond}>
   502→              <span className="material-symbols-outlined text-[14px]">add</span>Add condition</button>
   503→          </div>
   504→
   505→          {/* THEN */}
   506→          <div className="border-t border-surface-dim pt-4">
   507→            <div className="text-label-sm text-on-surface-variant uppercase tracking-wider font-bold mb-2">Then (apply)</div>
   508→            <div className="space-y-3">
   509→              <div>
   510→                <label className="field-label">Category</label>
   511→                <select className="field-input" data-then-category value={thenCategory} onChange={e => setThenCategory(e.target.value)}>
   512→                  <option value="">— Select category —</option>
   513→                  <option>Office supplies</option>
   514→                  <option>Shipping</option>
   515→                  <option>Software</option>
   516→                  <option>Advertising</option>
   517→                  <option>Meals &amp; entertainment</option>
   518→                  <option>Travel</option>

   521→                </select>
   522→              </div>
   523→              <div>
   524→                <label className="field-label">Apply to bank account</label>
   525→                <select className="field-input" value={bankAccount} onChange={e => setBankAccount(e.target.value)}>
   526→                  <option>All accounts</option>
   527→                  <option>Chase •9910</option>
   528→                  <option>Visa •4421</option>
   529→                  <option>Amex •2103</option>
   530→                </select>
   531→              </div>
   532→              <div className="flex items-center gap-2">
   533→                <input type="checkbox" className="row-check" checked={turnOnRule} onChange={e => setTurnOnRule(e.target.checked)} />
   534→                <span className="text-body-sm">Turn on rule</span>
   535→              </div>
   536→            </div>
   537→          </div>
   538→
   539→          {/* PREVIEW — which past transactions this rule would have matched */}
   540→          <div className="border-t border-surface-dim pt-4">
   541→            <div className="flex items-center justify-between mb-2">
   542→              <div className="text-label-sm text-on-surface-variant uppercase tracking-wider font-bold">Would have matched</div>
   543→              <span className="text-[11px] text-on-surface-variant"><span data-preview-count>{previewMatches.length}</span> recent transactions</span>
   544→            </div>
   545→            <div className="border border-surface-dim rounded-lg overflow-hidden" data-preview-list>
   546→              {!previewKw ? (
   547→                <div className="text-[11px] text-on-surface-variant text-center py-4">Enter a description keyword to preview matched transactions.</div>
   548→              ) : previewMatches.length === 0 ? (
   549→                <div className="text-[11px] text-on-surface-variant text-center py-4">No recent transactions match "<strong>{previewKw}</strong>".</div>
   550→              ) : (
   551→                previewMatches.slice(0, 6).map((t, i) => (
   552→                  <div className="preview-txn" key={i}>
   553→                    <span className="text-[11px] text-on-surface-variant w-20">{t.date}</span>
   554→                    <span className="flex-1 font-medium truncate">{t.desc}</span>
   555→                    <span className={`amt ${t.amt < 0 ? 'text-error' : 'text-secondary'}`}>{money(t.amt)}</span>
   556→                  </div>
   557→                ))
   558→              )}
   559→            </div>
   560→          </div>
   561→
   562→        </div>
   563→
   564→        <div className="h-14 border-t border-surface-dim px-4 flex items-center justify-between shrink-0">
   565→          <button className="h-8 px-3 border border-surface-dim rounded text-body-md hover:bg-surface-container-low" data-close-rule onClick={closeDrawer}>Cancel</button>
   566→          <div className="flex items-center gap-2">
   567→            <div className="flex split-btn">
   568→              <button className="h-8 px-3 bg-white border border-surface-dim text-body-md font-semibold hover:bg-surface-container-low split-btn-main" data-save-new onClick={() => saveRule(false)}>Save and new</button>
   569→              <button className="h-8 px-1.5 bg-white border border-surface-dim hover:bg-surface-container-low split-btn-arrow" aria-label="Save and new options">

   574→              <button className="h-8 px-1.5 bg-primary text-white hover:bg-primary-container split-btn-arrow" aria-label="Save rule options">
   575→                <span className="material-symbols-outlined text-[16px]">expand_more</span></button>
   576→            </div>
   577→          </div>
   578→        </div>
   579→      </div>
   580→
   581→      {/* ═══════════ ROW ⋯ MENU ═══════════ */}
   582→      {menu && menuRow && (
   583→        <div className="row-menu open" id="row-menu" style={{ top: menu.top, left: menu.left }}>
   584→          <div className="menu-header">This rule</div>
   585→          <button data-action="edit" onClick={() => handleMenuAction('edit', menuRow)}><span className="material-symbols-outlined">edit</span>Edit rule</button>
   586→          <button data-action="duplicate" onClick={() => handleMenuAction('duplicate', menuRow)}><span className="material-symbols-outlined">content_copy</span>Duplicate</button>
   587→          <div className="sep"></div>
   588→          <div className="menu-header">Run</div>
   589→          <button data-action="apply-existing" onClick={() => handleMenuAction('apply-existing', menuRow)}><span className="material-symbols-outlined">play_circle</span>Apply to existing</button>
   590→          <button data-action="view-matched" onClick={() => handleMenuAction('view-matched', menuRow)}><span className="material-symbols-outlined">list</span>View matched transactions</button>
   591→          <div className="sep"></div>
   592→          <div className="menu-header">State</div>
   593→          {!menu.paused && (
   594→            <button data-action="pause" onClick={() => handleMenuAction('pause', menuRow)}><span className="material-symbols-outlined">pause</span>Pause rule</button>
   595→          )}
   596→          {menu.paused && (
   597→            <button data-action="activate" data-hide="true" onClick={() => handleMenuAction('activate', menuRow)}><span className="material-symbols-outlined">play_arrow</span>Activate rule</button>
   598→          )}

   600→        </div>
   601→      )}
   602→
   603→      <div className={toastMsg ? 'toast show' : 'toast'} id="toast">{toastMsg}</div>
   604→    </div>
   605→  );
