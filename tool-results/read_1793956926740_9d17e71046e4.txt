    86→    86→// Maps a GroupBgToken to a concrete background-color class for child-pill icons.
    87→    87→function groupBgClass(bg: GroupBgToken): string {
    88→    88→  switch (bg) {
    89→    89→    case 'bg-qb-navy':
    90→    90→      return 'bg-[#1f2937]';
    91→    91→    case 'bg-qb-teal':
    92→    92→      return 'bg-[#0e5e6f]';
    93→    93→    case 'bg-qb-maroon':
    94→    94→      return 'bg-[#6b1e28]';
    95→    95→    case 'bg-qb-green':
    96→    96→      return 'bg-[#1e5a38]';
    97→    97→    case 'bg-error':
    98→    98→      return 'bg-[#ba1a1a]';
    99→    99→    case 'bg-primary':
   100→   100→      return 'bg-[#0a6400]';
   101→   101→    default:
   102→   102→      return 'bg-[#5a6355]';
   103→   103→  }
