   108→// Maps a GroupBgToken to a concrete background-color class for child-pill icons.
   109→function groupBgClass(bg: GroupBgToken): string {
   110→  switch (bg) {
   111→    case 'bg-qb-navy':
   112→      return 'bg-[#1f2937]';
   113→    case 'bg-qb-teal':
   114→      return 'bg-[#0e5e6f]';
   115→    case 'bg-qb-maroon':
   116→      return 'bg-[#6b1e28]';
   117→    case 'bg-qb-green':
   118→      return 'bg-[#1e5a38]';
   119→    case 'bg-error':
   120→      return 'bg-[#ba1a1a]';
   121→    case 'bg-primary':
   122→      return 'bg-[#0a6400]';
   123→    default:
   124→      return 'bg-[#5a6355]';
   125→  }
   126→}

   176→      tabMatch = String(r.status ?? '').toLowerCase() === 'for review';

   188→    setSelectedIds((prev) => {

   192→      return next;
   193→    });

   202→
   203→  const isStudioCatalog =

   206→

   232→                onNavigate('design', 'new-design');
   233→                onToast(`${row.product ?? 'Product'} added to designer.`);
   234→              }
   235→            },
   236→          },
   237→          {
   238→            label: 'View',
   239→            icon: 'open_in_new',
   240→            onClick: () => onOpenPeek(row, cfg.family),
   241→          },
   242→        ]
   243→      : [
   244→          {
   245→            label: 'View',
   246→            icon: 'open_in_new',
   247→            onClick: () => onOpenPeek(row, cfg.family),
   248→          },
   249→        ]),
   250→    {
   251→      label: 'Edit',
   252→      icon: 'edit',
   253→      onClick: () => onOpenPeek(row, cfg.family),
   254→    },
   255→    {
   256→      label: 'Duplicate',
   257→      icon: 'content_copy',
   258→      onClick: () => onToast(`Duplicated ${cfg.newLabel}`),
   259→    },
   260→    {
   261→      label: 'Print',
   262→      icon: 'print',
   263→      dividerBefore: true,
   264→      onClick: () => {
   265→        if (typeof window !== 'undefined') window.print();
   266→      },
   267→    },
   268→    {
   269→      label: 'Delete',
   270→      icon: 'delete',
   271→      danger: true,
   272→      onClick: () => {
   273→        onToast(`Deleted record`);
   274→        toggleSelect(idx);
   275→      },
   276→    },
   277→  ];
   278→
   279→  const bulkActions: import('@/components/ListPrimitives').BulkAction[] = [
   280→    {
   281→      label: 'Export selected',
   282→      icon: 'download',

   284→        const cols = cfg.cols.map((c) => ({ key: c.key, label: c.label }));
   285→        const selectedRows = selectedArray
   286→          .map((i) => filtered[i])
   287→          .filter(Boolean) as unknown as Record<string, unknown>[];

   290→        setSelectedIds(new Set());
   291→      },
   292→    },
   293→    {
   294→      label: 'Delete selected',
   295→      icon: 'delete',
   296→      danger: true,
   297→      onClick: () => {
   298→        onToast(`Deleted ${selectedArray.length} records`);
   299→        setSelectedIds(new Set());
   300→      },
   301→    },
   302→  ];
   303→
   304→  // Per-family create options for the split-button dropdown
   305→  const familyCreates = cfg.createActionsRow
   306→    ? cfg.createActionsRow.map(([label]) => label)
   307→    : [cfg.newLabel];
   308→
   309→  return (
   310→    <div className={styles.screenContainer}>
   311→      <div className={styles.listHeader}>

   322→          <div className={styles.splitBtn} ref={splitRef} style={{ position: 'relative' }}>

   324→              type="button"
   325→              className={styles.splitBtnPrimaryMain}
   326→              data-new={cfg.newLabel}
   327→              onClick={() => onCreate(cfg.newLabel)}

   329→              <span className={`material-symbols-outlined ${styles.icon16}`}>add</span>

   331→            </button>
   332→            <button
   333→              type="button"
   334→              className={styles.splitBtnPrimaryArrow}

   336→              onClick={() => setSplitOpen((prev) => !prev)}
   337→            >
   338→              <span className={`material-symbols-outlined ${styles.icon16}`}>expand_more</span>
   339→            </button>
   340→            {splitOpen && (
   341→              <div className={listStyles.splitDropdown} role="menu">
   342→                {familyCreates.map((label) => (
   343→                  <button
   344→                    key={label}
   345→                    type="button"
   346→                    className={listStyles.splitDropdownItem}
   347→                    onClick={() => {
   348→                      setSplitOpen(false);
   349→                      onCreate(label);
   350→                    }}
   351→                  >
   352→                    <span className="material-symbols-outlined" style={{ fontSize: 16 }}>add</span>
   353→                    {label}
   354→                  </button>
   355→                ))}
   356→                <div className={listStyles.rowMenuDivider} />
   357→                <button
   358→                  type="button"
   359→                  className={listStyles.splitDropdownItem}
   360→                  onClick={() => {
   361→                    setSplitOpen(false);
   362→                    onOpenCreateFlyout();
   363→                  }}
   364→                >
   365→                  <span className="material-symbols-outlined" style={{ fontSize: 16 }}>apps</span>
   366→                  Show all create options

   368→              </div>

   372→            type="button"
   373→            className={styles.secondaryBtn}

   376→              const cols = cfg.cols.map((c) => ({ key: c.key, label: c.label }));

   379→            }}
   380→          >
   381→            <span className={`material-symbols-outlined ${styles.icon16}`}>download</span>
   382→            Export
   383→          </button>
   384→          <button
   385→            type="button"
   386→            className={styles.secondaryBtn}
   387→            data-customize
   388→            onClick={() => onToast('Customize panel — Data / Visual tabs')}
   389→          >
   390→            <span className={`material-symbols-outlined ${styles.icon16}`}>tune</span>
   391→            Customize
   392→          </button>
   393→        </div>
   394→      </div>
   395→
   396→      {cfg.createActionsRow && (
   397→        <div className={styles.createActionsRow}>
   398→          <span className={styles.createActionsLabel}>Create actions:</span>
   399→          {cfg.createActionsRow.map(([label, icon]) => (
   400→            <button
   401→              key={label}
   402→              type="button"
   403→              className={styles.createActionChip}
   404→              data-new={label}
   405→              onClick={() => onCreate(label)}
   406→            >
   407→              <span className={`material-symbols-outlined ${styles.icon16}`}>{icon}</span>
   408→              {label}
   409→            </button>
   410→          ))}
   411→          <button
   412→            type="button"
   413→            className={styles.showAllBtn}
   414→            data-open-create
   415→            onClick={onOpenCreateFlyout}
   416→          >
   417→            <span className={`material-symbols-outlined ${styles.icon16}`}>apps</span>
   418→            Show all
   419→          </button>
   420→        </div>
   421→      )}
   422→
   423→      <div className={styles.tableCard}>
   424→        <div className={styles.viewTabsBar}>
   425→          {tabs.map((tab) => {
   426→            const isTabActive = activeTab === tab;
   427→            const count =
   428→              tab === 'All'
   429→                ? allRows.length
   430→                : allRows.filter(
   431→                    (r) => String(r.status ?? '').toLowerCase() === tab.toLowerCase()
   432→                  ).length;
   433→            return (
   434→              <button
   435→                key={tab}
   436→                type="button"
   437→                data-viewtab={tab}
   438→                className={`${styles.viewTab} ${isTabActive ? styles.viewTabActive : ''}`}

   441→                {tab}

   443→              </button>
   444→            );
   445→          })}
   446→        </div>
   447→
   448→        <div className={styles.filterToolbar}>
   449→          <div className={styles.rowFilterWrap}>
   450→            <span className={`material-symbols-outlined ${styles.icon15Search}`}>search</span>
   451→            <input
   452→              className={styles.rowFilterInput}
   453→              placeholder="Filter rows…"
   454→              type="text"
   455→              value={rowSearch}
   456→              onChange={(e) => setRowSearch(e.target.value)}

   459→          <button type="button" className={styles.filterToolBtn}>
   460→            <span className={`material-symbols-outlined ${styles.icon14}`}>filter_list</span>
   461→            Filter
   462→          </button>
   463→          <button type="button" className={styles.filterToolBtn}>
   464→            <span className={`material-symbols-outlined ${styles.icon14}`}>calendar_today</span>

   468→            <span className={`material-symbols-outlined ${styles.icon14}`}>bookmark</span>
   469→            Saved views
   470→          </button>
   471→          <span className={styles.rowCountText}>
   472→            {selectedIds.size > 0
   473→              ? `${selectedIds.size} selected`
   474→              : `1–${filtered.length} of ${filtered.length}`}
   475→          </span>
   476→        </div>
   477→
   478→        {selectedIds.size > 0 && (
   479→          <BulkBar
   480→            selectedIds={selectedArray.map(String)}
   481→            onClear={() => setSelectedIds(new Set())}
   482→            actions={bulkActions}
   483→          />
   484→        )}
   485→
   486→        {hasData ? (
   487→          <table className={styles.dataTable}>
   488→            <thead className={styles.tableHead}>
   489→              <tr>
   490→                <th className={styles.checkColHeader}>
   491→                  <input
   492→                    type="checkbox"
   493→                    className={styles.rowCheck}
   494→                    checked={selectedIds.size === filtered.length && filtered.length > 0}
   495→                    onChange={toggleSelectAll}
   496→                  />
   497→                </th>
   498→                {isStudioCatalog && <th className={styles.thumbColHeader} />}
   499→                {cfg.cols.map((col) => (
   500→                  <th
   501→                    key={col.key}
   502→                    className={`${styles.colHeader} ${col.right ? styles.alignRight : ''}`}
   503→                  >
   504→                    {col.label}
   505→                  </th>
   506→                ))}
   507→                <th className={styles.actionColHeader} />
   508→              </tr>
   509→            </thead>
   510→            <tbody className={styles.tableBody}>
   511→              {filtered.map((row, idx) => {
   512→                const prod = isStudioCatalog ? productForRow(row) : undefined;
   513→                const thumbImg = prod
   514→                  ? 'data:image/svg+xml;utf8,<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 100 100"><rect width="100" height="100" fill="%23e8e8e8"/><path d="M30 30 L30 90 L45 90 L45 50 L50 40 L55 50 L55 90 L70 90 L70 30 Z" fill="%23fff" stroke="%23ccc"/></svg>'
   515→                  : null;
   516→                return (
   517→                <tr
   518→                  key={idx}
   519→                  className={`${styles.tableRow} ${selectedIds.has(idx) ? styles.tableRowSelected : ''}`}
   520→                  onClick={() => onOpenPeek(row, cfg.family)}
   521→                >
   522→                  <td className={styles.cell}>
   523→                    <input
   524→                      type="checkbox"
   525→                      className={styles.rowCheck}
   526→                      checked={selectedIds.has(idx)}
   527→                      onClick={(e) => e.stopPropagation()}

   531→                  {isStudioCatalog && (
   532→                    <td className={styles.thumbCell}>

   558→                        key={col.key}
   559→                        className={`${styles.cell} ${col.right ? styles.alignRight : ''}`}

   566→                      </td>
   567→                    );

   627→function OverviewView({

   651→    const dom = DOMAINS[domain];

   654→      g.screens.map((s) => ({

   667→

   669→    const el = stripRef.current;
   670→    if (!el) return;
   671→    setCanLeft(el.scrollLeft > 4);
   672→    setCanRight(el.scrollLeft + el.clientWidth < el.scrollWidth - 4);
   673→  }, []);

   675→  useEffect(() => {
   676→    updateArrows();
   677→  }, [childScreens, updateArrows]);
   678→
   679→  const scrollBy = (dir: 1 | -1) => {
   680→    const el = stripRef.current;
   681→    if (!el) return;
   682→    el.scrollBy({ left: dir * Math.max(el.clientWidth * 0.7, 200), behavior: 'smooth' });
   683→  };

   686→    <div className={styles.screenContainer}>

   711→            >
   712→              <span className="material-symbols-outlined" style={{ fontSize: 18 }}>
   713→                chevron_right
   714→              </span>
   715→            </button>
   716→          )}

   718→            ref={stripRef}
   719→            className={styles.childStrip}
   720→            onScroll={updateArrows}
   721→          >

   740→                </button>

   742→            })}
   743→          </div>
   744→        </div>
   745→      )}

   750→          <span className={styles.breadcrumbSep}>/</span>
   751→          <span>{groupName}</span>
   752→        </div>
   753→        <h1 className={styles.pageTitle}>{cfg.title}</h1>
