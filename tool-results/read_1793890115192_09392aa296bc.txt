   379→function updateSummary(){
   380→  const slips = docType === "labels" ? 0 : selectedOrders;
   381→  const labels = docType === "slips" ? 0 : (selectedOrders > 0 ? Math.max(0, selectedOrders - (document.querySelector("[data-ord]").checked ? 0 : 1)) : 0);
   382→  // Simpler: labels only for Ship orders (pickup has no label). Of the 3 default, 2 are ship.
   383→  const shipOrders = [...document.querySelectorAll("[data-ord]")].filter(c=>c.checked && !c.closest(".ord-item").textContent.includes("Local pickup")).length;
   384→  const realLabels = docType === "slips" ? 0 : shipOrders;
   385→  const realSlips = docType === "labels" ? 0 : selectedOrders;
   386→
   387→  document.querySelector("[data-batch-orders]").textContent = selectedOrders;
   388→  document.querySelector("[data-batch-slips]").textContent = realSlips;
   389→  document.querySelector("[data-batch-labels]").textContent = realLabels;
   390→  const pages = template === "letter-4x6" ? Math.ceil((realSlips + realLabels) / 4) : Math.max(realSlips, realLabels);
   391→  document.querySelector("[data-batch-pages]").textContent = Math.max(1, pages);
   392→  document.querySelector("[data-preview-count]").textContent = Math.max(1, pages);
   393→}

   395→function updatePreview(){
   396→  const stage = document.querySelector("[data-preview-stage]");
   397→  if (selectedOrders === 0){
   398→    stage.innerHTML = `<div class="text-center text-white opacity-70 text-body-sm">No orders selected</div>`;
   399→    return;
   400→  }
   401→  if (docType === "labels" || (docType === "both" && template === "thermal-4x6")){
   402→    stage.innerHTML = renderLabel();
   403→  } else if (docType === "slips"){
   404→    stage.innerHTML = renderSlip();
   405→  } else {
   406→    // both — show slip (the label is on the other page in the real PDF)
   407→    stage.innerHTML = renderSlip() + `<div class="text-white text-[10px] mt-3 opacity-70">+ ${selectedOrders} shipping label${selectedOrders!==1?"s":""} on next page${selectedOrders!==1?"s":""}</div>`;
   408→  }
   409→}
