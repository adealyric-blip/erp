    12→export interface VisualFormatConfig {
    13→  currencySymbol: boolean;
    14→  parenthesesNegatives: boolean;

    26→export function formatVisualNumber(
    27→  amount: number,
    28→  cfg: VisualFormatConfig,
    29→  forcePlain?: boolean
    30→): string {
    31→  if (forcePlain) {
    32→    return amount.toLocaleString();
    33→  }
    34→  const scaled = cfg.divideBy1000 ? amount / 1000 : amount;
    35→  const absVal = Math.abs(scaled);
    36→  const formatted = absVal.toLocaleString('en-US', {
    37→    minimumFractionDigits: cfg.decimals,
    38→    maximumFractionDigits: cfg.decimals,
    39→  });
    40→  const suffix = cfg.divideBy1000 ? 'K' : '';
    41→  const prefix = cfg.currencySymbol ? '$' : '';
    42→  if (scaled < 0) {
    43→    return cfg.parenthesesNegatives
    44→      ? `(${prefix}${formatted}${suffix})`
    45→      : `−${prefix}${formatted}${suffix}`;
    46→  }
    47→  return `${prefix}${formatted}${suffix}`;
    48→}

    50→/* 1. Company Setup Wizard Modal (§G Settings -> Company setup wizard) */
    51→export function CompanySetupWizardModal({
    52→  steps,
    53→  initialStepIndex,
    54→  onSaveSteps,
    55→  onClose,
    56→  onToast,
    57→}: {
    58→  steps: SetupStepItem[];
    59→  initialStepIndex: number;
    60→  onSaveSteps: (updated: SetupStepItem[]) => void;
    61→  onClose: () => void;
    62→  onToast: (msg: string) => void;
    63→}) {
    64→  const [draftSteps, setDraftSteps] = useState<SetupStepItem[]>(steps);
    65→  const [activeIdx, setActiveIdx] = useState<number>(initialStepIndex);
    66→
    67→  const current = draftSteps[activeIdx] || draftSteps[0];
    68→
    69→  const handleFieldChange = (fieldKey: string, val: string) => {
    70→    setDraftSteps((prev) =>
    71→      prev.map((step, idx) =>
    72→        idx === activeIdx
    73→          ? {
    74→              ...step,
    75→              fields: step.fields.map((f) =>
    76→                f.key === fieldKey ? { ...f, value: val } : f
    77→              ),
    78→            }
    79→          : step
    80→      )
    81→    );
    82→  };
    83→
    84→  const handleCompleteStep = (andAdvance: boolean) => {
    85→    const nextSteps = draftSteps.map((step, idx) =>
    86→      idx === activeIdx ? { ...step, completed: true } : step
    87→    );
    88→    setDraftSteps(nextSteps);
    89→    onSaveSteps(nextSteps);
    90→    onToast(`Saved "${current.title}"`);
    91→    if (andAdvance && activeIdx < nextSteps.length - 1) {
    92→      setActiveIdx(activeIdx + 1);
    93→    } else if (!andAdvance) {
    94→      onClose();
    95→    }
    96→  };
    97→
    98→  const handleToggleCompleted = () => {
    99→    const nextSteps = draftSteps.map((step, idx) =>
   100→      idx === activeIdx ? { ...step, completed: !step.completed } : step
   101→    );
   102→    setDraftSteps(nextSteps);
   103→    onSaveSteps(nextSteps);
   104→  };
   105→
   106→  return (
   107→    <div className={styles.scrim} onClick={onClose}>
   108→      <div className={styles.modalBoxWide} onClick={(e) => e.stopPropagation()}>
   109→        <div className={styles.overlayHeader}>
   110→          <div>
   111→            <div className={styles.overlayTitle}>Company Setup Wizard</div>
   112→            <div className={styles.overlaySubtitle}>
   113→              Business profile → Connect bank → Payments → Branding → Operating goals
   114→            </div>
   115→          </div>
   116→          <button type="button" className={styles.closeIconBtn} onClick={onClose}>
   117→            <span className={`material-symbols-outlined ${styles.icon18}`}>close</span>
   118→          </button>
   119→        </div>
   120→
   121→        <div className={styles.wizardStepTabs}>
   122→          {draftSteps.map((st, idx) => (
   123→            <button
   124→              key={st.id}
   125→              type="button"
   126→              className={`${styles.wizardTabBtn} ${
   127→                idx === activeIdx ? styles.wizardTabBtnActive : ''
   128→              }`}
   129→              onClick={() => setActiveIdx(idx)}
   130→            >
   131→              <span className={`material-symbols-outlined ${styles.icon14}`}>
   132→                {st.completed ? 'check_circle' : st.icon}
   133→              </span>
   134→              {st.stepNumber}. {st.title}
   135→            </button>
   136→          ))}
   137→        </div>
   138→
   139→        <div className={styles.overlayBody}>
   140→          <div className={styles.panelHeader}>
   141→            <div>
   142→              <div className={styles.panelTitle}>
   143→                Step {current.stepNumber}: {current.title}
   144→              </div>
   145→              <div className={styles.panelSubtitle}>{current.subtitle}</div>
   146→            </div>
   147→            <button
   148→              type="button"
   149→              className={styles.btnGhostSm}
   150→              onClick={handleToggleCompleted}
   151→            >
   152→              <span className={`material-symbols-outlined ${styles.icon14}`}>
   153→                {current.completed ? 'Restart_alt' : 'check'}
   154→              </span>
   155→              {current.completed ? 'Mark incomplete' : 'Mark complete'}
   156→            </button>
   157→          </div>
   158→
   159→          {current.fields.map((field) => (
   160→            <div key={field.key} className={styles.formGroup}>
   161→              <label className={styles.formLabel}>{field.label}</label>
   162→              <input
   163→                type="text"
   164→                className={styles.formInput}
   165→                value={field.value}
   166→                onChange={(e) => handleFieldChange(field.key, e.target.value)}
   167→              />
   168→              {field.hint ? <span className={styles.formHint}>{field.hint}</span> : null}
   169→            </div>
   170→          ))}
   171→        </div>
   172→
   173→        <div className={styles.overlayFooter}>
   174→          <button type="button" className={styles.btnSecondary} onClick={onClose}>
   175→            Cancel
   176→          </button>
   177→          <div className={styles.footerRightGroup}>
   178→            <button
   179→              type="button"
   180→              className={styles.btnSecondary}
   181→              onClick={() => handleCompleteStep(false)}
   182→            >
   183→              Save and close
   184→            </button>
   185→            <button
   186→              type="button"
   187→              className={styles.btnPrimary}
   188→              onClick={() => handleCompleteStep(true)}
   189→            >
   190→              {activeIdx < draftSteps.length - 1 ? 'Complete & next step' : 'Finish setup'}
   191→            </button>
   192→          </div>
   193→        </div>
   194→      </div>
   195→    </div>
   196→  );
   197→}
