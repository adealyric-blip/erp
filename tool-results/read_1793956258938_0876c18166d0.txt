    40→    40→    domain,

    43→    43→  };
    44→    44→}

    80→    80→function loadPinMode(): PinMode {
    81→    81→  if (typeof window !== 'undefined') {
    82→    82→    try {
    83→    83→      const saved = localStorage.getItem(PIN_MODE_STORAGE_KEY);
    84→    84→      if (saved === 'bookmarks' || saved === 'favorites') return saved;
    85→    85→    } catch {
    86→    86→      // ignore
    87→    87→    }
    88→    88→  }
    89→    89→  return 'bookmarks';
    90→    90→}
    91→    91→
    92→    92→function loadBookmarks(): BookmarkEntry[] {
    93→    93→  if (typeof window !== 'undefined') {
    94→    94→    try {
    95→    95→      const saved = localStorage.getItem(BOOKMARKS_STORAGE_KEY);
    96→    96→      if (saved) {
    97→    97→        const parsed = JSON.parse(saved);
    98→    98→        if (Array.isArray(parsed)) {
    99→    99→          return parsed.filter(
   100→   100→            (b): b is BookmarkEntry =>
   101→   101→              b &&
   102→   102→              typeof b.domain === 'string' &&
   103→   103→              typeof b.screen === 'string' &&
   104→   104→              !!DOMAINS[b.domain as DomainKey]
   105→   105→          );
   106→   106→        }
   107→   107→      }
   108→   108→    } catch {
   109→   109→      // ignore
   110→   110→    }
   111→   111→  }
   112→   112→  return DEFAULT_BOOKMARKS;
   113→   113→}
