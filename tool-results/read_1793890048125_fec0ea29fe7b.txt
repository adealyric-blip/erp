   792→function applyFilters(){
   793→  const q = document.querySelector("[data-filter]").value.toLowerCase();
   794→  let shown = 0, total = 0;
   795→  document.querySelectorAll(".o-row").forEach(tr=>{
   796→    total++;
   797→    const tabOk = currentTab === "All" || tr.dataset.status === currentTab;
   798→    const qOk = !q || tr.textContent.toLowerCase().includes(q);
   799→    const vis = tabOk && qOk;
   800→    tr.style.display = vis ? "" : "none";
   801→    if (vis) shown++;
   802→  });
   803→  document.querySelector("[data-o-count]").textContent = `${shown} of ${total}`;
   804→}
