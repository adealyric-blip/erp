   629→      pill.textContent = "Sent";

   631→    case "remind": toast(`Reminder sent for ${id}`); break;
   632→    case "print": toast(`Printing statement ${id}`); break;
   633→    case "mark-paid":

   641→    case "delete":
   642→      tr.remove();

   644→      toast(`Statement ${id} deleted`); break;

   656→function applyFilters(){
   657→  const q = document.querySelector("[data-filter]").value.toLowerCase();
   658→  let shown = 0, total = 0;
   659→  document.querySelectorAll(".s-row").forEach(tr=>{
   660→    total++;
   661→    const tabOk = currentTab === "All" || tr.dataset.status === currentTab;
   662→    const qOk = !q || tr.textContent.toLowerCase().includes(q);
   663→    const vis = tabOk && qOk;
   664→    tr.style.display = vis ? "" : "none";
   665→    if (vis) shown++;
   666→  });
   667→  document.querySelector("[data-s-count]").textContent = `${shown} of ${total}`;
   668→}
