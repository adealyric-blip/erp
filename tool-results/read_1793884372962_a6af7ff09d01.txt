   340→   340→function OverviewView({

   348→   348→  onNavigate,
   349→   349→}: {
   350→   350→  cfg: OverviewScreenConfig;
   351→   351→  domain: DomainKey;
   352→   352→  domainLabel: string;
   353→   353→  groupName: string;
   354→   354→  activeScreenId: string;
   355→   355→  onCreate: (label: string) => void;
   356→   356→  onOpenCreateFlyout: () => void;
   357→   357→  onNavigate: (domain: DomainKey, screenId: string) => void;
   358→   358→}) {
   359→   359→  const kpis = cfg.kpis || [];
   360→   360→  const actions = cfg.actions || [];
   361→   361→
   362→   362→  // Build the child strip: all screens across all groups in this domain.
   363→   363→  const childScreens = useMemo(() => {
   364→   364→    const dom = DOMAINS[domain];
   365→   365→    if (!dom) return [];
   366→   366→    return dom.groups.flatMap((g) =>
   367→   367→      g.screens.map((s) => ({
   368→   368→        ...s,
   369→   369→        group: g.name,
   370→   370→        groupIcon: g.icon,
   371→   371→        groupBg: g.bg,
   372→   372→      }))
   373→   373→    );
   374→   374→  }, [domain]);

   376→   376→  // Hover-arrows navigation for the child strip (replaces scrollbar).

   382→   382→    const el = stripRef.current;

   385→   385→    setCanRight(el.scrollLeft + el.clientWidth < el.scrollWidth - 4);
   386→   386→  }, []);
   387→   387→
   388→   388→  useEffect(() => {
   389→   389→    updateArrows();
   390→   390→  }, [childScreens, updateArrows]);
   391→   391→
   392→   392→  const scrollBy = (dir: 1 | -1) => {
   393→   393→    const el = stripRef.current;
   394→   394→    if (!el) return;
   395→   395→    el.scrollBy({ left: dir * Math.max(el.clientWidth * 0.7, 200), behavior: 'smooth' });
   396→   396→  };
   397→   397→
   398→   398→  return (
   399→   399→    <div className={styles.screenContainer}>
   400→   400→      {/* Domain child strip — hover-arrow navigation (no scrollbar) */}
   401→   401→      {childScreens.length > 1 && (
   402→   402→        <div
   403→   403→          className={styles.childStripWrap}
   404→   404→          onMouseEnter={updateArrows}

   406→   406→          {canLeft && (
   407→   407→            <button
   408→   408→              type="button"
   409→   409→              className={`${styles.childStripArrow} ${styles.childStripArrowLeft}`}
   410→   410→              onClick={() => scrollBy(-1)}
   411→   411→              aria-label="Scroll left"
   412→   412→            >
   413→   413→              <span className="material-symbols-outlined" style={{ fontSize: 18 }}>
   414→   414→                chevron_left
   415→   415→              </span>
   416→   416→            </button>
   417→   417→          )}
   418→   418→          {canRight && (
   419→   419→            <button
   420→   420→              type="button"
   421→   421→              className={`${styles.childStripArrow} ${styles.childStripArrowRight}`}
   422→   422→              onClick={() => scrollBy(1)}
   423→   423→              aria-label="Scroll right"
   424→   424→            >
   425→   425→              <span className="material-symbols-outlined" style={{ fontSize: 18 }}>
   426→   426→                chevron_right
   427→   427→              </span>
   428→   428→            </button>
   429→   429→          )}
   430→   430→          <div
   431→   431→            ref={stripRef}
   432→   432→            className={styles.childStrip}
   433→   433→            onScroll={updateArrows}
   434→   434→          >
   435→   435→            {childScreens.map((s) => {
   436→   436→              const active = s.id === activeScreenId;
   437→   437→              return (
   438→   438→                <button
   439→   439→                  key={s.id}
   440→   440→                  type="button"
   441→   441→                  className={`${styles.childPill} ${active ? styles.childPillActive : ''}`}
   442→   442→                  onClick={() => onNavigate(domain, s.id)}
   443→   443→                  title={s.label}
   444→   444→                >
   445→   445→                  <span
   446→   446→                    className={`${styles.childPillIcon} ${groupBgClass(s.groupBg)}`}
   447→   447→                  >
   448→   448→                    <span className="material-symbols-outlined" style={{ fontSize: 13 }}>
   449→   449→                      {s.groupIcon}
   450→   450→                    </span>
   451→   451→                  </span>
   452→   452→                  {s.label}
   453→   453→                </button>
   454→   454→              );
   455→   455→            })}
   456→   456→          </div>
   457→   457→        </div>
   458→   458→      )}
   459→   459→
   460→   460→      <div className={styles.overviewHeader}>
   461→   461→        <div className={styles.breadcrumb}>
   462→   462→          <span>{domainLabel}</span>
   463→   463→          <span className={styles.breadcrumbSep}>/</span>
   464→   464→          <span>{groupName}</span>
   465→   465→        </div>
   466→   466→        <h1 className={styles.pageTitle}>{cfg.title}</h1>
   467→   467→        <p className={styles.pageSubtitle}>{cfg.subtitle}</p>
   468→   468→      </div>
   469→   469→
   470→   470→      <div className={styles.kpiGrid}>
   471→   471→        {kpis.map((kpi) => (
   472→   472→          <div key={kpi.label} className={styles.kpiCard}>
   473→   473→            <span className={styles.kpiLabel}>{kpi.label}</span>

   480→   480→              {kpi.delta || ''}

   482→   482→          </div>
   483→   483→        ))}
   484→   484→      </div>
   485→   485→
   486→   486→      <div className={styles.createActionsRow}>
   487→   487→        <span className={styles.createActionsLabel}>Create actions:</span>
   488→   488→        {actions.map(([label, icon]) => (
   489→   489→          <button
   490→   490→            key={label}
   491→   491→            type="button"
   492→   492→            className={styles.createActionChip}
   493→   493→            data-new={label}
   494→   494→            onClick={() => onCreate(label)}
   495→   495→          >
   496→   496→            <span className={`material-symbols-outlined ${styles.icon16}`}>{icon}</span>
