   730→function collectPayload(){

   732→  selectedChannels.forEach(n=>{
   733→    const el=document.getElementById("override-"+n.replace(/\W/g,''));

   735→  });
   736→  return {
   737→    title:document.getElementById("postTitle").value||"Untitled post",
   738→    caption:document.getElementById("postCaption").value,
   739→    link:document.getElementById("postLink").value,
   740→    hashtags:document.getElementById("postTags").value,
   741→    channels:selectedChannels,
   742→    channelOverrides:overrides,

   745→    extraAttachments,
   746→    publishMode,
   747→    date:publishMode==="schedule"?document.getElementById("scheduleDate").value:null,
   748→    time:publishMode==="schedule"?document.getElementById("scheduleTime").value:null,
   749→    timezone:document.getElementById("timezone").value,
   750→    approval:document.getElementById("approval").value
   751→  };
