   697→   697→function openDecision(action, ids){
   698→   698→  if (!ids || !ids.length){ toast("Select at least one item"); return; }
   699→   699→  activeDecision = {action, ids};
   700→   700→  const titles = {
   701→   701→    approve:"Approve content",
   702→   702→    changes:"Request changes",
   703→   703→    reject:"Reject content",
   704→   704→    assign:"Assign reviewer"
   705→   705→  };
   706→   706→  const subs = {
   707→   707→    approve:"This approves the content and unlocks publishing for the selected items.",
   708→   708→    changes:"The author will be notified and must resubmit. Existing schedule is paused.",
   709→   709→    reject:"The content is rejected and removed from the schedule. Logged for audit.",
   710→   710→    assign:"Reassign the reviewer. The previous reviewer will see the change in their queue."
   711→   711→  };
   712→   712→  document.getElementById("decisionTitle").textContent = titles[action];
   713→   713→  document.getElementById("decisionSub").textContent = subs[action];
   714→   714→  document.getElementById("decisionFields").classList.toggle("hidden-imp", action==="assign");
   715→   715→  document.getElementById("decisionReason").value="";
   716→   716→  document.getElementById("decisionNote").value="";
   717→   717→  document.getElementById("decisionConfirm").textContent = titles[action].split(" ")[0];
   718→   718→  document.getElementById("decisionModal").classList.add("open");
   719→   719→}
   720→   720→document.getElementById("decisionCancel").addEventListener("click",()=>{
   721→   721→  document.getElementById("decisionModal").classList.remove("open");
   722→   722→  activeDecision={action:null,ids:[]};
   723→   723→});
   724→   724→document.getElementById("decisionConfirm").addEventListener("click",async ()=>{
   725→   725→  const action=activeDecision.action, ids=activeDecision.ids.slice();
   726→   726→  const reason=document.getElementById("decisionReason").value;
   727→   727→  const reviewer=document.getElementById("decisionReviewer").value;
   728→   728→  const note=document.getElementById("decisionNote").value;
   729→   729→
   730→   730→  if (action!=="assign" && !reason){ toast("Pick a reason before confirming"); return; }
   731→   731→
   732→   732→  if (action==="assign"){
   733→   733→    await API.assign(ids[0],{reviewer});
   734→   734→    ids.forEach(id=>{ const it=items.find(x=>x.id===id); if(it) it.reviewer=reviewer; });
   735→   735→    toast("Assigned to "+reviewer);
   736→   736→  } else {
   737→   737→    const state = action==="approve"?"approved":action==="changes"?"changes":"rejected";
   738→   738→    await API.decide(ids.join(","),{decision:action,reason,reviewer,note});
   739→   739→    ids.forEach(id=>{
   740→   740→      const it=items.find(x=>x.id===id);
   741→   741→      if (it){ it.state=state; it.reviewer=reviewer; }
   742→   742→      decisions.unshift({when:new Date().toISOString(),title:(it&&it.title)||id,decision:state,reviewer,reason,note:note||"—"});
   743→   743→    });
   744→   744→    toast(action==="approve"?"Approved":action==="changes"?"Changes requested":"Rejected");
   745→   745→  }
   746→   746→  document.getElementById("decisionModal").classList.remove("open");
   747→   747→  activeDecision={action:null,ids:[]};
   748→   748→  renderQueue();
   749→   749→});
