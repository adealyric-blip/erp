   465→   465→function applyFilters(){
   466→   466→  const q = (document.querySelector("[data-filter-text]").value||"").toLowerCase();
   467→   467→  const kind = document.querySelector("[data-filter-mapped]").value;
   468→   468→  const tab = document.querySelector("[data-viewtab].active")?.dataset.viewtab || "all";
   469→   469→  let shown = 0, total = 0;
   470→   470→  document.querySelectorAll("[data-tax-tbody] tr").forEach(tr=>{
   471→   471→    if (tr.classList.contains("section")) return;
   472→   472→    total++;
   473→   473→    const name = (tr.querySelector("td")?.textContent||"").toLowerCase();
   474→   474→    const trKind = tr.dataset.kind || "";
   475→   475→    const trCat = tr.dataset.category || "";
   476→   476→    let visible = true;
   477→   477→    if (q && !name.includes(q)) visible = false;
   478→   478→    if (kind === "mapped" && trKind !== "mapped") visible = false;
   479→   479→    if (kind === "unmapped" && trKind !== "unmapped") visible = false;
   480→   480→    if (tab === "unmapped" && trKind !== "unmapped") visible = false;
   481→   481→    if (tab === "income" && trCat !== "income") visible = false;
   482→   482→    if (tab === "deductions" && trCat !== "deductions") visible = false;
   483→   483→    if (tab === "assets" && trCat !== "assets") visible = false;
   484→   484→    tr.style.display = visible ? "" : "none";
   485→   485→    if (visible) shown++;
   486→   486→  });
   487→   487→  // Sections: hide if empty
   488→   488→  document.querySelectorAll("[data-tax-tbody] tr.section").forEach(section=>{
   489→   489→    let next = section.nextElementSibling;
   490→   490→    let anyVisible = false;
   491→   491→    while (next && !next.classList.contains("section") && !next.classList.contains("final")){
   492→   492→      if (!next.classList.contains("total") && next.style.display !== "none" && next.dataset.kind) anyVisible = true;
   493→   493→      next = next.nextElementSibling;
   494→   494→    }
   495→   495→    section.style.display = anyVisible ? "" : "none";
   496→   496→    // Hide totals if no visible rows in section
   497→   497→    let row = section.nextElementSibling;
   498→   498→    while (row && !row.classList.contains("section") && !row.classList.contains("final")){
   499→   499→      if (row.classList.contains("total")) row.style.display = anyVisible ? "" : "none";
   500→   500→      row = row.nextElementSibling;
   501→   501→    }
   502→   502→  });
   503→   503→  document.querySelector("[data-tx-count]").textContent = `${shown} of ${total}`;
   504→   504→}
