  1214→/* auto stream simulation */
  1215→function pushSyntheticEvent(){
  1216→  const live = PIXELS.filter(p=>p.status === "Live");
  1217→  if (!live.length) return;
  1218→  const p = live[Math.floor(Math.random() * live.length)];
  1219→  const ev = p.events[Math.floor(Math.random() * p.events.length)];
  1220→  const value = ev.valueField ? Math.round((10 + Math.random() * 90) * 100) / 100 : null;
  1221→  const isPurchase = /purchase|completepayment|add_to_cart|addtocart/i.test(ev.name);
  1222→  appendStreamEvent({ at: nowIso(), pixel: p.id, event: ev.name, value, ok: true });
  1223→  p.events24 += 1; p.events30 += 1; p.lastEvent = nowIso();
  1224→  if (isPurchase) toast(`${p.name} fired ${ev.name}${value != null ? " · $" + value.toFixed(2) : ""}`);
  1225→  renderTable();
  1226→}
