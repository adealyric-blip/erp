   680→function syncSelectAll(){
   681→  const all = document.querySelector("[data-check-all]");

   687→function syncBulk(){
   688→  const n = [...document.querySelectorAll(".cr-check")].filter(cb=>cb.checked).length;
   689→  const bar = document.querySelector("[data-bulk-bar]");
   690→  bar.classList.toggle("show", n > 0);
   691→  document.querySelector("[data-bulk-count]").textContent = `${n} selected`;
   692→}
   693→document.addEventListener("change", e=>{
   694→  if (e.target.classList.contains("cr-check")) syncBulk();
   695→  if (e.target.matches("[data-check-all]")){
   696→    const on = e.target.checked;
   697→    document.querySelectorAll(".cr-check").forEach(cb=>{
   698→      if (cb.closest("tr").style.display !== "none") cb.checked = on;
   699→    });
   700→    syncBulk();
   701→  }
   702→});
   703→document.querySelectorAll("[data-bulk]").forEach(b=>b.addEventListener("click", ()=>{
   704→  const act = b.dataset.bulk;
   705→  if (act === "close"){
   706→    document.querySelectorAll(".cr-check").forEach(cb=>cb.checked = false);
   707→    document.querySelector("[data-check-all]").checked = false;
   708→    syncBulk(); return;
   709→  }
   710→  const selected = [...document.querySelectorAll(".cr-check:checked")].map(cb=>cb.closest("tr").dataset.row);
   711→  const n = selected.length;
   712→  if (act === "publish"){ selected.forEach(id=>{ const c = CREATIVES.find(x=>x.id===id); if (c) c.status = "In review"; }); toast(`Submitted ${n} creative(s)`); }
   713→  else if (act === "archive"){ selected.forEach(id=>{ const c = CREATIVES.find(x=>x.id===id); if (c) c.status = "Archived"; }); toast(`Archived ${n} creative(s)`); }
   714→  else if (act === "duplicate"){
   715→    selected.forEach(id=>{
   716→      const c = CREATIVES.find(x=>x.id===id);
   717→      if (c) CREATIVES.unshift({...c, id:"CR-" + String(CREATIVES.length + 1).padStart(3,"0"), name:c.name + " (copy)", status:"Draft", usedIn:0, impressions:0, ctr:0, roas:0});
   718→    });
   719→    toast(`Duplicated ${n} creative(s)`);
   720→  }
   721→  else if (act === "delete"){
   722→    if (confirm(`Delete ${n} creative(s)?`)){
   723→      selected.forEach(id=>{
   724→        const idx = CREATIVES.findIndex(x=>x.id===id);
   725→        if (idx >= 0) CREATIVES.splice(idx, 1);
   726→      });
   727→      toast(`Deleted ${n} creative(s)`);
   728→    }
   729→  }
   730→  renderTable();
   731→  document.querySelectorAll(".cr-check").forEach(cb=>cb.checked = false);
   732→  document.querySelector("[data-check-all]").checked = false;
   733→  syncBulk();
   734→}));
