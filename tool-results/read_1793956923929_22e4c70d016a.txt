    86→// Maps a GroupBgToken to a concrete background-color class for child-pill icons.
    87→function groupBgClass(bg: GroupBgToken): string {
    88→  switch (bg) {
    89→    case 'bg-qb-navy':
    90→      return 'bg-[#1f2937]';
    91→    case 'bg-qb-teal':
    92→      return 'bg-[#0e5e6f]';
    93→    case 'bg-qb-maroon':
    94→      return 'bg-[#6b1e28]';
    95→    case 'bg-qb-green':
    96→      return 'bg-[#1e5a38]';
    97→    case 'bg-error':
    98→      return 'bg-[#ba1a1a]';
    99→    case 'bg-primary':
   100→      return 'bg-[#0a6400]';
   101→    default:
   102→      return 'bg-[#5a6355]';
   103→  }
   104→}

   594→function OverviewView({
   595→  cfg,
   596→  domain,
   597→  domainLabel,
   598→  groupName,
   599→  activeScreenId,
   600→  onCreate,
   601→  onOpenCreateFlyout,
   602→  onNavigate,
   603→}: {
   604→  cfg: OverviewScreenConfig;
   605→  domain: DomainKey;
   606→  domainLabel: string;
   607→  groupName: string;
   608→  activeScreenId: string;
   609→  onCreate: (label: string) => void;
   610→  onOpenCreateFlyout: () => void;
   611→  onNavigate: (domain: DomainKey, screenId: string) => void;
   612→}) {
   613→  const kpis = cfg.kpis || [];
   614→  const actions = cfg.actions || [];
   615→
   616→  // Build the child strip: all screens across all groups in this domain.
   617→  const childScreens = useMemo(() => {
   618→    const dom = DOMAINS[domain];
   619→    if (!dom) return [];
   620→    return dom.groups.flatMap((g) =>
   621→      g.screens.map((s) => ({
   622→        ...s,
   623→        group: g.name,
   624→        groupIcon: g.icon,
   625→        groupBg: g.bg,
   626→      }))
   627→    );
   628→  }, [domain]);
   629→
   630→  // Hover-arrows navigation for the child strip (replaces scrollbar).
   631→  const stripRef = useRef<HTMLDivElement | null>(null);
   632→  const [canLeft, setCanLeft] = useState(false);
   633→  const [canRight, setCanRight] = useState(false);
   634→
   635→  const updateArrows = useCallback(() => {
   636→    const el = stripRef.current;
   637→    if (!el) return;
   638→    setCanLeft(el.scrollLeft > 4);
   639→    setCanRight(el.scrollLeft + el.clientWidth < el.scrollWidth - 4);
   640→  }, []);
   641→
   642→  useEffect(() => {
   643→    updateArrows();
   644→  }, [childScreens, updateArrows]);
   645→
   646→  const scrollBy = (dir: 1 | -1) => {
   647→    const el = stripRef.current;
   648→    if (!el) return;
   649→    el.scrollBy({ left: dir * Math.max(el.clientWidth * 0.7, 200), behavior: 'smooth' });
   650→  };
   651→
   652→  return (
   653→    <div className={styles.screenContainer}>
   654→      {/* Domain child strip — hover-arrow navigation (no scrollbar) */}
   655→      {childScreens.length > 1 && (
   656→        <div
   657→          className={styles.childStripWrap}
   658→          onMouseEnter={updateArrows}
   659→        >
   660→          {canLeft && (
   661→            <button
   662→              type="button"
   663→              className={`${styles.childStripArrow} ${styles.childStripArrowLeft}`}
   664→              onClick={() => scrollBy(-1)}
   665→              aria-label="Scroll left"
   666→            >
   667→              <span className="material-symbols-outlined" style={{ fontSize: 18 }}>
   668→                chevron_left
   669→              </span>
   670→            </button>
   671→          )}
   672→          {canRight && (
   673→            <button
   674→              type="button"
   675→              className={`${styles.childStripArrow} ${styles.childStripArrowRight}`}
   676→              onClick={() => scrollBy(1)}
   677→              aria-label="Scroll right"
   678→            >
   679→              <span className="material-symbols-outlined" style={{ fontSize: 18 }}>
   680→                chevron_right
   681→              </span>
   682→            </button>
   683→          )}
   684→          <div
   685→            ref={stripRef}
   686→            className={styles.childStrip}
   687→            onScroll={updateArrows}
   688→          >
   689→            {childScreens.map((s) => {
   690→              const active = s.id === activeScreenId;
   691→              return (
   692→                <button
   693→                  key={s.id}
   694→                  type="button"
   695→                  className={`${styles.childPill} ${active ? styles.childPillActive : ''}`}
   696→                  onClick={() => onNavigate(domain, s.id)}
   697→                  title={s.label}
   698→                >
   699→                  <span
   700→                    className={`${styles.childPillIcon} ${groupBgClass(s.groupBg)}`}
   701→                  >
   702→                    <span className="material-symbols-outlined" style={{ fontSize: 13 }}>
   703→                      {s.groupIcon}
   704→                    </span>
   705→                  </span>
   706→                  {s.label}
   707→                </button>
   708→              );
   709→            })}
   710→          </div>
   711→        </div>
   712→      )}
   713→
   714→      <div className={styles.overviewHeader}>
   715→        <div className={styles.breadcrumb}>
   716→          <span>{domainLabel}</span>
   717→          <span className={styles.breadcrumbSep}>/</span>
   718→          <span>{groupName}</span>
   719→        </div>
   720→        <h1 className={styles.pageTitle}>{cfg.title}</h1>
   721→        <p className={styles.pageSubtitle}>{cfg.subtitle}</p>
   722→      </div>
   723→
   724→      <div className={styles.kpiGrid}>
   725→        {kpis.map((kpi) => (
   726→          <div key={kpi.label} className={styles.kpiCard}>
   727→            <span className={styles.kpiLabel}>{kpi.label}</span>
   728→            <div className={styles.kpiValue}>{kpi.value}</div>
   729→            <div
   730→              className={
   731→                kpi.delta?.startsWith('+') ? styles.kpiDeltaPos : styles.kpiDeltaNeg
   732→              }
   733→            >
   734→              {kpi.delta || ''}
   735→            </div>
   736→          </div>
   737→        ))}
   738→      </div>
   739→
   740→      <div className={styles.createActionsRow}>
   741→        <span className={styles.createActionsLabel}>Create actions:</span>
   742→        {actions.map(([label, icon]) => (
   743→          <button
   744→            key={label}
   745→            type="button"
   746→            className={styles.createActionChip}
   747→            data-new={label}
   748→            onClick={() => onCreate(label)}
   749→          >
   750→            <span className={`material-symbols-outlined ${styles.icon16}`}>{icon}</span>
   751→            {label}
   752→          </button>
   753→        ))}
   754→        <button
   755→          type="button"
   756→          className={styles.showAllBtn}
   757→          data-open-create
   758→          onClick={onOpenCreateFlyout}
   759→        >
   760→          <span className={`material-symbols-outlined ${styles.icon16}`}>apps</span>
   761→          Show all
   762→        </button>
   763→      </div>
   764→
   765→      <div className={styles.panelCard}>
   766→        <h2 className={styles.panelTitle}>{cfg.panelTitle || 'Business at a glance'}</h2>
   767→        <p className={styles.panelText}>
   768→          {cfg.panelText ||
   769→            'Insight cards and suggestions live here. Setup checklist card until company setup is complete.'}
   770→        </p>
   771→      </div>
   772→    </div>
   773→  );
