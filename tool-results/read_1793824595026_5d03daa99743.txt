   655→   655→function renderReportBody(r){
   656→   656→  const body = document.querySelector("[data-report-body]");
   657→   657→  // headers depend on shape (some reports have 2 cols, some 3, some 5)
   658→   658→  const maxCells = Math.max(...r.rows.map(row => row.cells.length));
   659→   659→  // Build a header row for section kind if it has non-empty strings (some use it as header)
   660→   660→  let html = "";
   661→   661→  r.rows.forEach(row=>{
   662→   662→    if (row.kind === "section"){
   663→   663→      const cells = row.cells.map(c => `<th>${c}</th>`).join("");
   664→   664→      // if section cells look like a header (all string, no number), render as header row
   665→   665→      html += `<tr class="section">${cells}</tr>`;
   666→   666→    } else if (row.kind === "sub"){
   667→   667→      html += `<tr class="sub">${row.cells.map((c,i)=>`<td class="${i===0?"":"right"}">${formatNeg(c)}</td>`).join("")}</tr>`;
   668→   668→    } else if (row.kind === "total"){
   669→   669→      const label = row.label || "";
   670→   670→      const cells = row.cells.map((c,i)=>`<td class="right">${formatNeg(c)}</td>`).join("");
   671→   671→      html += `<tr class="total"><td>${label}</td>${cells}</tr>`;
   672→   672→    } else if (row.kind === "final"){
   673→   673→      const label = row.label || "";
   674→   674→      const cells = row.cells.map((c,i)=>`<td class="right">${formatNeg(c)}</td>`).join("");
   675→   675→      html += `<tr class="total" style="background:#f4f3f7"><td>${label}</td>${cells}</tr>`;
   676→   676→    }
   677→   677→  });
   678→   678→  body.innerHTML = html;
   679→   679→}
   680→   680→function formatNeg(v){
   681→   681→  if (!v || v === "—" || v === "") return v;
   682→   682→  if (negFormat === "red" && v.startsWith("−")) return `<span class="num-neg">${v}</span>`;
   683→   683→  if (negFormat === "paren" && v.startsWith("−")) return `<span class="num-neg">(${v.slice(1)})</span>`;
   684→   684→  return v;
   685→   685→}
