   373→   465→function applyFilters(){
   374→   466→  const q = (document.querySelector("[data-filter-text]").value||"").toLowerCase();
   375→   467→  const kind = document.querySelector("[data-filter-mapped]").value;
   376→   468→  const tab = document.querySelector("[data-viewtab].active")?.dataset.viewtab || "all";
   377→   469→  let shown = 0, total = 0;
   378→   470→  document.querySelectorAll("[data-tax-tbody] tr").forEach(tr=>{
   379→   471→    if (tr.classList.contains("section")) return;
   380→   472→    total++;
   381→   473→    const name = (tr.querySelector("td")?.textContent||"").toLowerCase();
   382→   474→    const trKind = tr.dataset.kind || "";
   383→   475→    const trCat = tr.dataset.category || "";
   384→   476→    let visible = true;
   385→   477→    if (q && !name.includes(q)) visible = false;
   386→   478→    if (kind === "mapped" && trKind !== "mapped") visible = false;
   387→   479→    if (kind === "unmapped" && trKind !== "unmapped") visible = false;
   388→   480→    if (tab === "unmapped" && trKind !== "unmapped") visible = false;
   389→   481→    if (tab === "income" && trCat !== "income") visible = false;
   390→   482→    if (tab === "deductions" && trCat !== "deductions") visible = false;
   391→   483→    if (tab === "assets" && trCat !== "assets") visible = false;
   392→   484→    tr.style.display = visible ? "" : "none";
   393→   485→    if (visible) shown++;
   394→   486→  });
   395→   487→  // Sections: hide if empty
   396→   488→  document.querySelectorAll("[data-tax-tbody] tr.section").forEach(section=>{
   397→   489→    let next = section.nextElementSibling;
   398→   490→    let anyVisible = false;
   399→   491→    while (next && !next.classList.contains("section") && !next.classList.contains("final")){
   400→   492→      if (!next.classList.contains("total") && next.style.display !== "none" && next.dataset.kind) anyVisible = true;
   401→   493→      next = next.nextElementSibling;
   402→   494→    }
   403→   495→    section.style.display = anyVisible ? "" : "none";
   404→   496→    // Hide totals if no visible rows in section
   405→   497→    let row = section.nextElementSibling;
   406→   498→    while (row && !row.classList.contains("section") && !row.classList.contains("final")){
   407→   499→      if (row.classList.contains("total")) row.style.display = anyVisible ? "" : "none";
   408→   500→      row = row.nextElementSibling;
   409→   501→    }
   410→   502→  });
   411→   503→  document.querySelector("[data-tx-count]").textContent = `${shown} of ${total}`;
   412→   504→}
   413→   505→document.querySelector("[data-filter-text]").addEventListener("input", applyFilters);
   414→   506→document.querySelector("[data-filter-mapped]").addEventListener("change", applyFilters);
