   187→      setAcctTax('');

   212→      /* Create new — insert at top of table */
   213→      const created: AccountRow = {
   214→        id: nextId.current++,
   215→        type,
   216→        name,
   217→        number,
   218→        detail,
   219→        detailText: detail,
   220→        tax: tax || '—',
   221→        balance: '$0.00',
   222→        bankbal: '—',
   223→        inactive: false,
   224→      };
   225→      setRows(rs => [created, ...rs]);
   226→      toast(`Account "${name}" created`);
   227→    }
   228→    if (andClose) {
   229→      closeAccountDrawer();

   245→  const handleNewAcctSplit = (e: React.MouseEvent<HTMLButtonElement>) => {

   264→    setEditMode(next);

   279→  };
   280→  const handleInlineDetailChange = (id: number, v: string) => {
   281→    patchRow(id, { detail: v });
   282→    setInlineDetailDisplay(prev => ({ ...prev, [id]: v }));
   283→  };
   284→

   287→    const name = row.name;
   288→    const bal = row.balance || '$0.00';

   381→
   382→  const menuRow = rowMenu ? rows.find(r => r.id === rowMenu.rowId) || null : null;
   383→
   384→  /* Edit-mode toggle button classes (mockup toggles !bg-qb-blue/!text-white/!border-qb-blue) */

   390→
   391→  return (
   392→    <div className="chart-of-accounts-page bg-background font-sans text-body-md text-on-surface antialiased">
   393→
   394→      {/* Context header (the mockup's "Preview · standalone" label is intentionally not rendered) */}
   395→      <div className="h-[52px] px-6 flex items-center justify-between bg-white border-b border-surface-dim">

   400→          <span className="text-label-md">Accounting</span>

   402→          <span className="text-label-md text-primary font-semibold">Chart of accounts</span>
   403→        </div>
   404→      </div>
   405→
   406→      <main className="px-6 pb-12">
   407→
   408→        {/* Page header */}
   409→        <div className="flex items-center justify-between gap-4 py-4">
   410→          <div>
   411→            <h1 className="text-headline-lg">Chart of accounts</h1>
   412→            <p className="text-body-sm text-on-surface-variant mt-0.5">34 accounts · Numbered, typed, mapped to tax lines.</p>
   413→          </div>
   414→          <div className="flex items-center gap-2">
   415→            <div className="flex split-btn">
   416→              <button className="h-8 px-3 bg-primary text-white text-body-md font-semibold hover:bg-primary-container flex items-center gap-1 split-btn-main" data-new-account onClick={() => openAccountDrawer(null)}>
   417→                <span className="material-symbols-outlined text-[16px]">add</span>New account
