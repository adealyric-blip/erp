   220→
   221→export default function ReportsScreen({ onNavigate, onToast }: ReportsScreenProps) {
   222→  const [activeView, setActiveView] = useState<'library' | 'viewer'>('library');
   223→  const [selectedReportId, setSelectedReportId] = useState<string>('pnl');
   224→  const [activeCategoryTab, setActiveCategoryTab] = useState<string>('all');
   225→  const [searchQuery, setSearchQuery] = useState('');
   226→
   227→  // Date range
   228→  const [dateFrom, setDateFrom] = useState('2026-11-01');
   229→  const [dateTo, setDateTo] = useState('2026-11-30');
   230→  const [rangePreset, setRangePreset] = useState('This month');
   231→
   232→  // Customize Drawer
   233→  const [isCustomizeOpen, setIsCustomizeOpen] = useState(false);
   234→  const [custTab, setCustTab] = useState<'data' | 'visual' | 'saved'>('data');
   235→  const [negFormat, setNegFormat] = useState<'red' | 'paren' | 'minus'>('red');
   236→  const [savedViewName, setSavedViewName] = useState('');
   237→
   238→  // Toast
   239→  const [toastMsg, setToastMsg] = useState<string | null>(null);
   240→  const timerRef = useRef<NodeJS.Timeout | null>(null);
   241→
   242→  const showToast = (msg: string) => {
   243→    if (onToast) onToast(msg);
   244→    setToastMsg(msg);
   245→    if (timerRef.current) clearTimeout(timerRef.current);
   246→    timerRef.current = setTimeout(() => setToastMsg(null), 2000);
   247→  };
   248→
   249→  useEffect(() => {
   250→    return () => {
   251→      if (timerRef.current) clearTimeout(timerRef.current);
   252→    };
   253→  }, []);
   254→
   255→  const currentReport = REPORTS[selectedReportId] || REPORTS.pnl;
   256→
   257→  // Filtered reports in Library
   258→  const filteredReports = useMemo(() => {
   259→    return Object.values(REPORTS).filter((r) => {
   260→      if (activeCategoryTab !== 'all' && r.category !== activeCategoryTab) return false;
   261→      if (searchQuery) {
   262→        const q = searchQuery.toLowerCase();
   263→        return r.title.toLowerCase().includes(q) || r.description.toLowerCase().includes(q);
   264→      }
   265→      return true;
   266→    });
   267→  }, [activeCategoryTab, searchQuery]);
   268→
   269→  const openReportViewer = (id: string) => {
   270→    setSelectedReportId(id);
   271→    setActiveView('viewer');

   273→  };
   274→
   275→  const handlePresetChange = (preset: string) => {
   276→    setRangePreset(preset);
   277→    if (preset === 'This month') {
   278→      setDateFrom('2026-11-01');
   279→      setDateTo('2026-11-30');
   280→    } else if (preset === 'Last month') {
   281→      setDateFrom('2026-10-01');
   282→      setDateTo('2026-10-31');
   283→    } else if (preset === 'This quarter') {
   284→      setDateFrom('2026-10-01');
   285→      setDateTo('2026-12-31');
   286→    } else if (preset === 'This year') {
   287→      setDateFrom('2026-01-01');
   288→      setDateTo('2026-12-31');
   289→    } else if (preset === 'Last year') {
   290→      setDateFrom('2025-01-01');
   291→      setDateTo('2025-12-31');
   292→    }
   293→  };
   294→
   295→  const formatNegative = (val: string) => {
   296→    if (!val || !val.startsWith('−')) return val;
   297→    if (negFormat === 'red') {
   298→      return <span className={styles.numNeg}>{val}</span>;
   299→    }
   300→    if (negFormat === 'paren') {
   301→      return <span className={styles.numNeg}>({val.slice(1)})</span>;
   302→    }
   303→    return val;
   304→  };
   305→
   306→  return (
   307→    <div className={styles.container}>

   317→          <span className={styles.breadcrumbSep}>/</span>
   318→          <span className={styles.breadcrumbActive}>
   319→            {activeView === 'viewer' ? currentReport.title : 'Reports'}

   326→      {/* VIEW 1 — REPORT LIBRARY                                    */}
   327→      {/* ═══════════════════════════════════════════════════════════ */}
   328→      {activeView === 'library' && (
   329→        <main className={styles.main}>
   330→          <div className={styles.pageHeader}>

   332→              <h1 className={styles.pageTitle}>Reports</h1>

   334→                Standard reports, custom reports, saved views.
   335→              </p>
   336→            </div>
   337→            <div className={styles.headerActions}>
   338→              <button
   339→                type="button"

   374→          {/* Category Tabs */}
