   382→export function screensOf(domain: DomainConfig): ScreenWithGroup[] {
   383→  return domain.groups.flatMap((g) => g.screens.map((s) => ({ ...s, group: g.name })));
   384→}

   386→export function findScreen(domainId: DomainKey, screenId: string): ScreenWithGroup {
   387→  const all = screensOf(DOMAINS[domainId]);
   388→  return all.find((s) => s.id === screenId) || all[0];
