    82→// Maps a GroupBgToken to a concrete background-color class for child-pill icons.
    83→function groupBgClass(bg: GroupBgToken): string {
    84→  switch (bg) {
    85→    case 'bg-qb-navy':
    86→      return 'bg-[#1f2937]';
    87→    case 'bg-qb-teal':
    88→      return 'bg-[#0e5e6f]';
    89→    case 'bg-qb-maroon':
    90→      return 'bg-[#6b1e28]';
    91→    case 'bg-qb-green':
    92→      return 'bg-[#1e5a38]';
    93→    case 'bg-error':
    94→      return 'bg-[#ba1a1a]';
    95→    case 'bg-primary':
    96→      return 'bg-[#0a6400]';
    97→    default:
    98→      return 'bg-[#5a6355]';
    99→  }

   421→                    {col.label}

   428→              {filtered.map((row, idx) => (

   430→                  key={idx}

   433→                >
   434→                  <td className={styles.cell}>
   435→                    <input
   436→                      type="checkbox"
   437→                      className={styles.rowCheck}

   443→                  {cfg.cols.map((col) => {
   444→                    const val = row[col.key];
   445→                    return (
   446→                      <td
   447→                        key={col.key}
   448→                        className={`${styles.cell} ${col.right ? styles.alignRight : ''}`}

   450→                        {col.key === 'status' ? (
   451→                          <StatusPill status={String(val)} />
   452→                        ) : (

   457→                  })}
   458→                  <td className={styles.actionCell}>
   459→                    <RowMenu items={rowMenuItems(row, idx)} />
   460→                  </td>
   461→                </tr>
   462→              ))}
   463→            </tbody>
   464→          </table>
   465→        ) : (
   466→          <div className={styles.emptyState}>
   467→            <div className={styles.emptyIconCircle}>
   468→              <span className={`material-symbols-outlined ${styles.icon32Outline}`}>inbox</span>
   469→            </div>
   470→            <h3 className={styles.emptyTitle}>
   471→              {cfg.emptyTitle || `No ${cfg.title.toLowerCase()} yet`}
   472→            </h3>
   473→            <p className={styles.emptyDesc}>
   474→              {cfg.emptySteps || 'Create your first record or import from a CSV.'}
   475→            </p>
   476→            <div className={styles.emptyActions}>
   477→              <button
   478→                type="button"
   479→                className={styles.secondaryBtn}
   480→                data-import
   481→                onClick={() => onToast('Import dialog — select a CSV file')}

   486→              <button

   493→                {cfg.newLabel}
   494→              </button>
   495→            </div>
   496→          </div>
   497→        )}
   498→      </div>
   499→    </div>

   501→}

   503→function OverviewView({
   504→  cfg,
   505→  domain,
   506→  domainLabel,
   507→  groupName,
   508→  activeScreenId,
   509→  onCreate,
   510→  onOpenCreateFlyout,
   511→  onNavigate,
   512→}: {
   513→  cfg: OverviewScreenConfig;
   514→  domain: DomainKey;
   515→  domainLabel: string;
   516→  groupName: string;
   517→  activeScreenId: string;
   518→  onCreate: (label: string) => void;
   519→  onOpenCreateFlyout: () => void;
   520→  onNavigate: (domain: DomainKey, screenId: string) => void;
   521→}) {
   522→  const kpis = cfg.kpis || [];
   523→  const actions = cfg.actions || [];
   524→
   525→  // Build the child strip: all screens across all groups in this domain.
   526→  const childScreens = useMemo(() => {
   527→    const dom = DOMAINS[domain];
   528→    if (!dom) return [];
   529→    return dom.groups.flatMap((g) =>
   530→      g.screens.map((s) => ({
   531→        ...s,
   532→        group: g.name,
   533→        groupIcon: g.icon,
   534→        groupBg: g.bg,
   535→      }))
   536→    );
   537→  }, [domain]);
   538→
   539→  // Hover-arrows navigation for the child strip (replaces scrollbar).
   540→  const stripRef = useRef<HTMLDivElement | null>(null);
   541→  const [canLeft, setCanLeft] = useState(false);
   542→  const [canRight, setCanRight] = useState(false);
   543→
   544→  const updateArrows = useCallback(() => {
   545→    const el = stripRef.current;
   546→    if (!el) return;
   547→    setCanLeft(el.scrollLeft > 4);
   548→    setCanRight(el.scrollLeft + el.clientWidth < el.scrollWidth - 4);
   549→  }, []);
   550→
   551→  useEffect(() => {
   552→    updateArrows();
   553→  }, [childScreens, updateArrows]);
   554→
   555→  const scrollBy = (dir: 1 | -1) => {
   556→    const el = stripRef.current;
   557→    if (!el) return;
   558→    el.scrollBy({ left: dir * Math.max(el.clientWidth * 0.7, 200), behavior: 'smooth' });
   559→  };
   560→
   561→  return (
   562→    <div className={styles.screenContainer}>
   563→      {/* Domain child strip — hover-arrow navigation (no scrollbar) */}
   564→      {childScreens.length > 1 && (
   565→        <div
   566→          className={styles.childStripWrap}
   567→          onMouseEnter={updateArrows}
   568→        >
   569→          {canLeft && (
   570→            <button
   571→              type="button"
   572→              className={`${styles.childStripArrow} ${styles.childStripArrowLeft}`}
   573→              onClick={() => scrollBy(-1)}
   574→              aria-label="Scroll left"
   575→            >
   576→              <span className="material-symbols-outlined" style={{ fontSize: 18 }}>
   577→                chevron_left
   578→              </span>
   579→            </button>
   580→          )}
   581→          {canRight && (
   582→            <button
   583→              type="button"
   584→              className={`${styles.childStripArrow} ${styles.childStripArrowRight}`}
   585→              onClick={() => scrollBy(1)}
   586→              aria-label="Scroll right"
   587→            >
   588→              <span className="material-symbols-outlined" style={{ fontSize: 18 }}>
   589→                chevron_right
   590→              </span>
   591→            </button>
   592→          )}
   593→          <div
   594→            ref={stripRef}
   595→            className={styles.childStrip}
   596→            onScroll={updateArrows}
   597→          >
   598→            {childScreens.map((s) => {
   599→              const active = s.id === activeScreenId;
   600→              return (
   601→                <button
   602→                  key={s.id}
   603→                  type="button"
