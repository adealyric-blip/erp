   321→const money = n => "$" + n.toLocaleString("en-US",{minimumFractionDigits:2,maximumFractionDigits:2});
   322→
   323→/* ═══════════════════════════════════════════════════════════ */
   324→/* State + selection + total                                  */
   325→/* ═══════════════════════════════════════════════════════════ */
   326→function updateTotals(){
   327→  const rows = [...document.querySelectorAll(".pay-row")];
   328→  let selected = 0, total = 0;
   329→  rows.forEach(tr=>{
   330→    const checked = tr.querySelector(".pay-check").checked;
   331→    tr.classList.toggle("checked", checked);
   332→    if (checked){
   333→      selected++;
   334→      total += parseFloat((tr.querySelector("[data-pay-amount]").value||"0").replace(/,/g,"")) || 0;
   335→    }
   336→  });
   337→  document.querySelector("[data-pay-total]").textContent = money(total);
   338→  document.querySelector("[data-pay-count]").textContent = `${selected} of ${rows.length} selected`;
   339→  document.querySelector("[data-selection-summary]").textContent = `${selected} bill${selected!==1?"s":""} selected`;
   340→  document.querySelector("[data-submit]").disabled = selected === 0;
   341→  document.querySelector("[data-submit-count]").textContent = selected;

   384→document.querySelector("[data-filter]").addEventListener("input", applyFilters);

   386→function applyFilters(){
   387→  const q = document.querySelector("[data-filter]").value.toLowerCase();
   388→  const due = document.querySelector("[data-due-filter]").value;
   389→  let shown = 0;
   390→  document.querySelectorAll(".pay-row").forEach(tr=>{
   391→    const qOk = !q || tr.dataset.vendor.toLowerCase().includes(q) || tr.dataset.billno.toLowerCase().includes(q);
   392→    let dueOk = true;
   393→    if (due !== "all"){
   394→      const dueDate = new Date(tr.dataset.due);
   395→      const days = Math.round((dueDate - today) / 86400000);
   396→      if (due === "7") dueOk = days >= 0 && days <= 7;
   397→      if (due === "30") dueOk = days >= 0 && days <= 30;
   398→      if (due === "overdue") dueOk = days < 0 || tr.dataset.overdue === "true";
   399→    }
   400→    const vis = qOk && dueOk;
   401→    tr.style.display = vis ? "" : "none";
   402→    if (vis) shown++;
   403→    // uncheck hidden
   404→    if (!vis && tr.querySelector(".pay-check").checked){
   405→      tr.querySelector(".pay-check").checked = false;
   406→    }
   407→  });
   408→  updateTotals();
   409→}
