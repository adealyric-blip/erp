   498→  if (!rowMenu.contains(e.target)) rowMenu.classList.remove("open");

   500→rowMenu.querySelectorAll("[data-action]").forEach(b=>b.addEventListener("click",()=>{
   501→  const a = b.dataset.action;
   502→  rowMenu.classList.remove("open");
   503→  const name = currentRow?.querySelector("td:nth-child(3)")?.textContent?.trim() || "product";
   504→  if (a === "edit") openDrawer(name);
   505→  else if (a === "delete") toast(`Delete "${name}"? Confirm in modal.`);
   506→  else toast(`${a.charAt(0).toUpperCase()+a.slice(1)} → ${name}`);
   507→}));

   510→function updateBulk(){
   511→  const checks = [...document.querySelectorAll(".prod-check")];
   512→  const checked = checks.filter(c=>c.checked).length;
   513→  const bar = document.querySelector("[data-bulk-bar]");
   514→  const cnt = document.querySelector("[data-bulk-count]");
   515→  if (checked>0) { bar.classList.add("show"); cnt.textContent = `${checked} selected`; }
   516→  else bar.classList.remove("show");
   517→}
