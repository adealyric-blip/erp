    67→// Maps a GroupBgToken to a concrete background-color class for child-pill icons.
    68→function groupBgClass(bg: GroupBgToken): string {
    69→  switch (bg) {
    70→    case 'bg-qb-navy':
    71→      return 'bg-[#1f2937]';
    72→    case 'bg-qb-teal':
    73→      return 'bg-[#0e5e6f]';
    74→    case 'bg-qb-maroon':
    75→      return 'bg-[#6b1e28]';
    76→    case 'bg-qb-green':
    77→      return 'bg-[#1e5a38]';
    78→    case 'bg-error':
    79→      return 'bg-[#ba1a1a]';
    80→    case 'bg-primary':
    81→      return 'bg-[#0a6400]';
    82→    default:
    83→      return 'bg-[#5a6355]';
    84→  }

   340→function OverviewView({
   341→  cfg,
   342→  domain,
   343→  domainLabel,
   344→  groupName,
   345→  activeScreenId,
   346→  onCreate,
   347→  onOpenCreateFlyout,
   348→  onNavigate,
   349→}: {
   350→  cfg: OverviewScreenConfig;
   351→  domain: DomainKey;
   352→  domainLabel: string;
   353→  groupName: string;
   354→  activeScreenId: string;
   355→  onCreate: (label: string) => void;
   356→  onOpenCreateFlyout: () => void;
   357→  onNavigate: (domain: DomainKey, screenId: string) => void;
   358→}) {
   359→  const kpis = cfg.kpis || [];
   360→  const actions = cfg.actions || [];
   361→
   362→  // Build the child strip: all screens across all groups in this domain.
   363→  const childScreens = useMemo(() => {
   364→    const dom = DOMAINS[domain];
   365→    if (!dom) return [];
   366→    return dom.groups.flatMap((g) =>
   367→      g.screens.map((s) => ({
   368→        ...s,
   369→        group: g.name,
   370→        groupIcon: g.icon,
   371→        groupBg: g.bg,
   372→      }))
   373→    );
   374→  }, [domain]);
   375→
   376→  // Hover-arrows navigation for the child strip (replaces scrollbar).
   377→  const stripRef = useRef<HTMLDivElement | null>(null);
   378→  const [canLeft, setCanLeft] = useState(false);
   379→  const [canRight, setCanRight] = useState(false);
   380→
   381→  const updateArrows = useCallback(() => {
   382→    const el = stripRef.current;
   383→    if (!el) return;
   384→    setCanLeft(el.scrollLeft > 4);
   385→    setCanRight(el.scrollLeft + el.clientWidth < el.scrollWidth - 4);
   386→  }, []);
   387→
   388→  useEffect(() => {
   389→    updateArrows();
   390→  }, [childScreens, updateArrows]);
   391→
   392→  const scrollBy = (dir: 1 | -1) => {
   393→    const el = stripRef.current;
   394→    if (!el) return;
   395→    el.scrollBy({ left: dir * Math.max(el.clientWidth * 0.7, 200), behavior: 'smooth' });
   396→  };
   397→
   398→  return (
   399→    <div className={styles.screenContainer}>
   400→      {/* Domain child strip — hover-arrow navigation (no scrollbar) */}
   401→      {childScreens.length > 1 && (
   402→        <div
   403→          className={styles.childStripWrap}
   404→          onMouseEnter={updateArrows}
   405→        >
   406→          {canLeft && (
   407→            <button
   408→              type="button"
   409→              className={`${styles.childStripArrow} ${styles.childStripArrowLeft}`}
   410→              onClick={() => scrollBy(-1)}
   411→              aria-label="Scroll left"
   412→            >
   413→              <span className="material-symbols-outlined" style={{ fontSize: 18 }}>
   414→                chevron_left
   415→              </span>
   416→            </button>
   417→          )}
   418→          {canRight && (
   419→            <button
   420→              type="button"
   421→              className={`${styles.childStripArrow} ${styles.childStripArrowRight}`}
   422→              onClick={() => scrollBy(1)}
   423→              aria-label="Scroll right"
   424→            >
   425→              <span className="material-symbols-outlined" style={{ fontSize: 18 }}>
   426→                chevron_right
   427→              </span>
   428→            </button>
   429→          )}
   430→          <div
   431→            ref={stripRef}
   432→            className={styles.childStrip}
   433→            onScroll={updateArrows}
   434→          >
   435→            {childScreens.map((s) => {
   436→              const active = s.id === activeScreenId;
   437→              return (
   438→                <button
   439→                  key={s.id}
   440→                  type="button"
   441→                  className={`${styles.childPill} ${active ? styles.childPillActive : ''}`}
   442→                  onClick={() => onNavigate(domain, s.id)}
   443→                  title={s.label}
   444→                >
   445→                  <span
   446→                    className={`${styles.childPillIcon} ${groupBgClass(s.groupBg)}`}
   447→                  >
   448→                    <span className="material-symbols-outlined" style={{ fontSize: 13 }}>
   449→                      {s.groupIcon}
   450→                    </span>
   451→                  </span>
   452→                  {s.label}
   453→                </button>
   454→              );
   455→            })}
   456→          </div>
   457→        </div>
   458→      )}
   459→
   460→      <div className={styles.overviewHeader}>
   461→        <div className={styles.breadcrumb}>
   462→          <span>{domainLabel}</span>
   463→          <span className={styles.breadcrumbSep}>/</span>
   464→          <span>{groupName}</span>
   465→        </div>
   466→        <h1 className={styles.pageTitle}>{cfg.title}</h1>
   467→        <p className={styles.pageSubtitle}>{cfg.subtitle}</p>
   468→      </div>
   469→
   470→      <div className={styles.kpiGrid}>
   471→        {kpis.map((kpi) => (
   472→          <div key={kpi.label} className={styles.kpiCard}>
   473→            <span className={styles.kpiLabel}>{kpi.label}</span>
   474→            <div className={styles.kpiValue}>{kpi.value}</div>
   475→            <div
   476→              className={
   477→                kpi.delta?.startsWith('+') ? styles.kpiDeltaPos : styles.kpiDeltaNeg
   478→              }
   479→            >
   480→              {kpi.delta || ''}
   481→            </div>
   482→          </div>
   483→        ))}
   484→      </div>
   485→
   486→      <div className={styles.createActionsRow}>
   487→        <span className={styles.createActionsLabel}>Create actions:</span>
   488→        {actions.map(([label, icon]) => (
   489→          <button
   490→            key={label}
   491→            type="button"
   492→            className={styles.createActionChip}
   493→            data-new={label}
   494→            onClick={() => onCreate(label)}
   495→          >
   496→            <span className={`material-symbols-outlined ${styles.icon16}`}>{icon}</span>
   497→            {label}
   498→          </button>
   499→        ))}
   500→        <button
   501→          type="button"
   502→          className={styles.showAllBtn}
   503→          data-open-create
   504→          onClick={onOpenCreateFlyout}
   505→        >
   506→          <span className={`material-symbols-outlined ${styles.icon16}`}>apps</span>
   507→          Show all
   508→        </button>
   509→      </div>
   510→
   511→      <div className={styles.panelCard}>
   512→        <h2 className={styles.panelTitle}>{cfg.panelTitle || 'Business at a glance'}</h2>
   513→        <p className={styles.panelText}>
   514→          {cfg.panelText ||
   515→            'Insight cards and suggestions live here. Setup checklist card until company setup is complete.'}
   516→        </p>
   517→      </div>
   518→    </div>
   519→  );
   520→}

   585→        onNavigate={onNavigate}

   587→        onCreate={onCreate}
   588→        onOpenCreateFlyout={onOpenCreateFlyout}

   591→  }

   596→        onNavigate={onNavigate}
   597→        onToast={onToast}
   598→        onCreate={onCreate}

   600→      />

   602→  }

   605→    return (
   606→      <PayBills
   607→        onNavigate={onNavigate}
   608→        onToast={onToast}

   610→        onOpenCreateFlyout={onOpenCreateFlyout}

   612→    );
   613→  }
   614→
   615→  if (screenId === 'vendors') {
   616→    return (
   617→      <Vendors
