   216→  const toastTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   217→  const toast = useCallback((m: string) => {

   219→    if (toastTimer.current) clearTimeout(toastTimer.current);
   220→    toastTimer.current = setTimeout(() => setToastMsg(''), 1800);

   226→  /* Library: filter + open */

   232→  /* Viewer: title / range text (two DOM spots, synced like the script) */

   239→  const [rangePreset, setRangePreset] = useState('Custom');
   240→
   241→  /* Customize drawer state */

   245→  const [rowsSel, setRowsSel] = useState('All accounts');

   252→  const [divideThousands, setDivideThousands] = useState(false);

   261→  const currentReport = currentReportId ? REPORTS[currentReportId] : null;

   276→  /* Negative-value formatting (red / paren), minus sign is U+2212 like the mockup */
   277→  const formatNeg = (v: string) => {
   278→    if (!v || v === '—' || v === '') return v;
   279→    if (negFormat === 'red' && v.startsWith('−')) return <span className="num-neg">{v}</span>;
   280→    if (negFormat === 'paren' && v.startsWith('−')) return <span className="num-neg">{`(${v.slice(1)})`}</span>;
   281→    return v;

   284→  /* Date range preset */
   285→  const handlePresetChange = (e: ChangeEvent<HTMLSelectElement>) => {
   286→    const preset = e.target.value;
   287→    setRangePreset(preset);
   288→    let nextFrom = rangeFrom;
   289→    let nextTo = rangeTo;
   290→    if (preset === 'This month') { nextFrom = '2026-11-01'; nextTo = '2026-11-30'; }

   292→    if (preset === 'This quarter') { nextFrom = '2026-10-01'; nextTo = '2026-12-31'; }
   293→    if (preset === 'This year') { nextFrom = '2026-01-01'; nextTo = '2026-12-31'; }
   294→    if (preset === 'Last year') { nextFrom = '2025-01-01'; nextTo = '2025-12-31'; }
   295→    setRangeFrom(nextFrom);
   296→    setRangeTo(nextTo);
   297→    const rangeText = `${nextFrom} – ${nextTo}`;
   298→    setViewerSub(rangeText);
   299→    setReportRange(rangeText);
   300→  };
   301→
   302→  const handleFromChange = (e: ChangeEvent<HTMLInputElement>) => {
   303→    const v = e.target.value;
   304→    setRangeFrom(v);
   305→    const rangeText = `${v} – ${rangeTo}`;
   306→    setViewerSub(rangeText);
   307→    setReportRange(rangeText);
   308→  };
   309→
   310→  const handleToChange = (e: ChangeEvent<HTMLInputElement>) => {
   311→    const v = e.target.value;
   312→    setRangeTo(v);
   313→    const rangeText = `${rangeFrom} – ${v}`;
   314→    setViewerSub(rangeText);
   315→    setReportRange(rangeText);
   316→  };
   317→
   318→  /* Column drag reorder */
   319→  const handleColDragStart = (col: string) => {
   320→    dragSrc.current = col;
   321→    setDraggingCol(col);
   322→  };
   323→  const handleColDragEnd = () => {
   324→    setDraggingCol(null);
   325→    dragSrc.current = null;
   326→  };
   327→  const handleColDragOver = (e: DragEvent<HTMLDivElement>, item: ColItem) => {
   328→    e.preventDefault();
   329→    if (!dragSrc.current || dragSrc.current === item.col) return;
   330→    const rect = e.currentTarget.getBoundingClientRect();
   331→    const after = (e.clientY - rect.top) > rect.height / 2;
   332→    setCols(prev => {
   333→      const moved = prev.find(c => c.col === dragSrc.current);
   334→      if (!moved) return prev;
   335→      const rest = prev.filter(c => c !== moved);
   336→      let insertAt = rest.indexOf(item);
   337→      if (insertAt === -1) return prev;
   338→      if (after) insertAt += 1;
   339→      rest.splice(insertAt, 0, moved);
   340→      return rest;
   341→    });
   342→  };
   343→  const toggleCol = (col: string) => {
   344→    setCols(prev => prev.map(c => (c.col === col ? { ...c, checked: !c.checked } : c)));
   345→  };
   346→
   347→  /* Save view */
   348→  const handleSaveView = () => {
   349→    const name = saveName.trim();
   350→    if (!name) { toast('Name required'); return; }
   351→    toast(`Saved view "${name}"`);
   352→  };
   353→
   354→  /* Apply customization (report body re-renders from state automatically) */
   355→  const handleApplyCust = () => {
   356→    setCustOpen(false);
   357→    toast('Customization applied');
   358→  };
   359→
   360→  /* Library filtering (search matches the card's semibold title, tab matches category) */
   361→  const q = reportSearch.toLowerCase();
   362→  const isCardVisible = (card: ReportCardDef) => {
   363→    const matchQ = !q || card.title.toLowerCase().includes(q);
   364→    const matchTab = rptTab === 'all' || card.category === rptTab;
   365→    return matchQ && matchTab;
   366→  };
   367→
   368→  return (
   369→    <div className="reports-page bg-background font-sans text-body-md text-on-surface antialiased">
   370→
   371→      {/* Context header */}
   372→      <div className="h-[52px] px-6 flex items-center justify-between bg-white border-b border-surface-dim">
   373→        <div className="flex items-center gap-3">
   374→          <span className="material-symbols-outlined text-[20px] text-on-surface-variant">account_balance</span>
   375→          <span className="text-label-md font-semibold">Finance</span>
   376→          <span className="text-outline">/</span>
   377→          <span className="text-label-md">Accounting</span>
   378→          <span className="text-outline">/</span>
   379→          <span className="text-label-md text-primary font-semibold" data-breadcrumb>{breadcrumb}</span>
   380→        </div>
   381→      </div>
   382→
   383→      {/* ═══════════════════════════════════════════════════════════ */}
   384→      {/* VIEW 1 — REPORT LIBRARY                                    */}
   385→      {/* ═══════════════════════════════════════════════════════════ */}
   386→      <main id="library-view" className={view === 'library' ? 'px-6 pb-12' : 'px-6 pb-12 hidden-imp'}>
   387→
   388→        <div className="flex items-center justify-between gap-4 py-4">
   389→          <div>
   390→            <h1 className="text-headline-lg">Reports</h1>
   391→            <p className="text-body-sm text-on-surface-variant mt-0.5">Standard reports, custom reports, saved views.</p>
   392→          </div>
   393→          <div className="flex items-center gap-2">
   394→            <button
