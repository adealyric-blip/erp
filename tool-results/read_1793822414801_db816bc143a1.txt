   745→function applyFilters(){
   746→  const q = document.querySelector("[data-filter]").value.toLowerCase();
   747→  let shown = 0, total = 0;
   748→  document.querySelectorAll(".e-row").forEach(tr=>{
   749→    total++;
   750→    const tabOk = currentTab === "All" || tr.dataset.status === currentTab;
   751→    const qOk = !q || tr.textContent.toLowerCase().includes(q);
   752→    const vis = tabOk && qOk;
   753→    tr.style.display = vis ? "" : "none";
   754→    if (vis) shown++;
   755→  });
   756→  document.querySelector("[data-e-count]").textContent = `${shown} of ${total}`;
   757→}
