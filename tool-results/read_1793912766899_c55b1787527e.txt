    81→// Maps a GroupBgToken to a concrete background-color class for child-pill icons.
    82→function groupBgClass(bg: GroupBgToken): string {
    83→  switch (bg) {
    84→    case 'bg-qb-navy':
    85→      return 'bg-[#1f2937]';
    86→    case 'bg-qb-teal':
    87→      return 'bg-[#0e5e6f]';
    88→    case 'bg-qb-maroon':
    89→      return 'bg-[#6b1e28]';
    90→    case 'bg-qb-green':
    91→      return 'bg-[#1e5a38]';
    92→    case 'bg-error':
    93→      return 'bg-[#ba1a1a]';
    94→    case 'bg-primary':
    95→      return 'bg-[#0a6400]';
    96→    default:
    97→      return 'bg-[#5a6355]';
    98→  }

   444→                    return (
   445→                      <td
   446→                        key={col.key}

   450→                          <StatusPill status={String(val)} />

   462→            </tbody>
   463→          </table>
   464→        ) : (
   465→          <div className={styles.emptyState}>
   466→            <div className={styles.emptyIconCircle}>
   467→              <span className={`material-symbols-outlined ${styles.icon32Outline}`}>inbox</span>
   468→            </div>
   469→            <h3 className={styles.emptyTitle}>
   470→              {cfg.emptyTitle || `No ${cfg.title.toLowerCase()} yet`}
   471→            </h3>
   472→            <p className={styles.emptyDesc}>
   473→              {cfg.emptySteps || 'Create your first record or import from a CSV.'}
   474→            </p>
   475→            <div className={styles.emptyActions}>
   476→              <button
   477→                type="button"
   478→                className={styles.secondaryBtn}

   502→function OverviewView({
   503→  cfg,
   504→  domain,
   505→  domainLabel,
   506→  groupName,
   507→  activeScreenId,
   508→  onCreate,
   509→  onOpenCreateFlyout,
   510→  onNavigate,
   511→}: {
   512→  cfg: OverviewScreenConfig;
   513→  domain: DomainKey;
   514→  domainLabel: string;
   515→  groupName: string;
   516→  activeScreenId: string;
   517→  onCreate: (label: string) => void;
   518→  onOpenCreateFlyout: () => void;
   519→  onNavigate: (domain: DomainKey, screenId: string) => void;
   520→}) {
   521→  const kpis = cfg.kpis || [];
   522→  const actions = cfg.actions || [];
   523→
   524→  // Build the child strip: all screens across all groups in this domain.
   525→  const childScreens = useMemo(() => {
   526→    const dom = DOMAINS[domain];
   527→    if (!dom) return [];
   528→    return dom.groups.flatMap((g) =>
   529→      g.screens.map((s) => ({
   530→        ...s,
   531→        group: g.name,
   532→        groupIcon: g.icon,
   533→        groupBg: g.bg,
   534→      }))
   535→    );
   536→  }, [domain]);
   537→
   538→  // Hover-arrows navigation for the child strip (replaces scrollbar).
   539→  const stripRef = useRef<HTMLDivElement | null>(null);
   540→  const [canLeft, setCanLeft] = useState(false);
   541→  const [canRight, setCanRight] = useState(false);
   542→
   543→  const updateArrows = useCallback(() => {
   544→    const el = stripRef.current;
   545→    if (!el) return;
   546→    setCanLeft(el.scrollLeft > 4);
   547→    setCanRight(el.scrollLeft + el.clientWidth < el.scrollWidth - 4);
   548→  }, []);
   549→
   550→  useEffect(() => {
   551→    updateArrows();
   552→  }, [childScreens, updateArrows]);
   553→
   554→  const scrollBy = (dir: 1 | -1) => {
   555→    const el = stripRef.current;
   556→    if (!el) return;
   557→    el.scrollBy({ left: dir * Math.max(el.clientWidth * 0.7, 200), behavior: 'smooth' });
   558→  };
   559→
   560→  return (
   561→    <div className={styles.screenContainer}>
   562→      {/* Domain child strip — hover-arrow navigation (no scrollbar) */}
   563→      {childScreens.length > 1 && (
   564→        <div
   565→          className={styles.childStripWrap}
   566→          onMouseEnter={updateArrows}
   567→        >
   568→          {canLeft && (
   569→            <button
   570→              type="button"
   571→              className={`${styles.childStripArrow} ${styles.childStripArrowLeft}`}
   572→              onClick={() => scrollBy(-1)}
   573→              aria-label="Scroll left"
   574→            >
   575→              <span className="material-symbols-outlined" style={{ fontSize: 18 }}>
   576→                chevron_left
   577→              </span>
   578→            </button>
   579→          )}
   580→          {canRight && (
   581→            <button
   582→              type="button"
   583→              className={`${styles.childStripArrow} ${styles.childStripArrowRight}`}
   584→              onClick={() => scrollBy(1)}
   585→              aria-label="Scroll right"
   586→            >
   587→              <span className="material-symbols-outlined" style={{ fontSize: 18 }}>
   588→                chevron_right
   589→              </span>
   590→            </button>
   591→          )}
   592→          <div
   593→            ref={stripRef}
   594→            className={styles.childStrip}
   595→            onScroll={updateArrows}
   596→          >
   597→            {childScreens.map((s) => {
   598→              const active = s.id === activeScreenId;
   599→              return (
   600→                <button
   601→                  key={s.id}
   602→                  type="button"
   603→                  className={`${styles.childPill} ${active ? styles.childPillActive : ''}`}
