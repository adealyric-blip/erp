   726→export function screensOf(domain: DomainConfig): ScreenWithGroup[] {
   727→  return domain.groups.flatMap((g) => g.screens.map((s) => ({ ...s, group: g.name })));
   728→}
   729→
   730→export function findScreen(domainId: DomainKey, screenId: string): ScreenWithGroup {
   731→  const all = screensOf(DOMAINS[domainId]);
   732→  return all.find((s) => s.id === screenId) || all[0];
   733→}
