   140→interface EstimatesScreenProps {
   141→  onCreateInvoice?: () => void;
   142→  onCreateCustomer?: () => void;
   143→  onOpenCreateFlyout?: () => void;
   144→  onToast?: (message: string) => void;

   146→
   147→export default function EstimatesScreen({
   148→  onCreateInvoice,
   149→  onCreateCustomer,
   150→  onOpenCreateFlyout,
   151→  onToast,
   152→}: EstimatesScreenProps) {
   153→  const [estimates, setEstimates] = useState<EstimateItem[]>(INITIAL_ESTIMATES);
   154→  const [activeTab, setActiveTab] = useState<string>('All');
   155→  const [searchFilter, setSearchFilter] = useState<string>('');
   156→  const [selectedIds, setSelectedIds] = useState<string[]>([]);
   157→
   158→  // Row context menu state
   159→  const [menuOpen, setMenuOpen] = useState(false);
   160→  const [menuPos, setMenuPos] = useState({ top: 0, left: 0 });
   161→  const [activeMenuEst, setActiveMenuEst] = useState<EstimateItem | null>(null);
   162→
   163→  // New estimate split dropdown
   164→  const [newMenuOpen, setNewMenuOpen] = useState(false);
   165→  const [newMenuPos, setNewMenuPos] = useState({ top: 0, left: 0 });
   166→  const newSplitBtnRef = useRef<HTMLButtonElement | null>(null);
   167→
   168→  // Drawer state
   169→  const [drawerOpen, setDrawerOpen] = useState(false);
   170→  const [editingEst, setEditingEst] = useState<EstimateItem | null>(null);
   171→
   172→  // Convert confirmation modal
   173→  const [convertModalOpen, setConvertModalOpen] = useState(false);
   174→  const [convertTargetEst, setConvertTargetEst] = useState<EstimateItem | null>(null);
   175→
   176→  // Close menus on click outside
   177→  useEffect(() => {
   178→    const handleGlobalClick = () => {
   179→      setMenuOpen(false);
   180→      setNewMenuOpen(false);
   181→    };
   182→    document.addEventListener('click', handleGlobalClick);
   183→    return () => document.removeEventListener('click', handleGlobalClick);
   184→  }, []);
   185→
   186→  // Filtered estimates list
   187→  const filteredEstimates = useMemo(() => {
   188→    const q = searchFilter.trim().toLowerCase();
   189→    return estimates.filter((est) => {
   190→      const tabMatch =
   191→        activeTab === 'All' ||
   192→        (activeTab === 'Converted' ? est.status === 'Converted' : est.status === activeTab);
   193→      if (!tabMatch) return false;
   194→      if (!q) return true;
   195→      return (
   196→        est.no.toLowerCase().includes(q) ||
   197→        est.customer.toLowerCase().includes(q) ||
   198→        est.email.toLowerCase().includes(q) ||
   199→        est.status.toLowerCase().includes(q)
   200→      );
   201→    });
   202→  }, [estimates, activeTab, searchFilter]);
   203→

   205→  const handleCheckAll = (e: React.ChangeEvent<HTMLInputElement>) => {
   206→    if (e.target.checked) {
   207→      setSelectedIds(filteredEstimates.map((e) => e.id));
   208→    } else {
   209→      setSelectedIds([]);
   210→    }
   211→  };
   212→
   213→  const handleToggleCheck = (id: string, e: React.MouseEvent) => {
   214→    e.stopPropagation();
   215→    setSelectedIds((prev) =>
   216→      prev.includes(id) ? prev.filter((i) => i !== id) : [...prev, id]
   217→    );
   218→  };
   219→
   220→  const handleBulkAction = (action: string) => {
   221→    const count = selectedIds.length;
   222→    if (action === 'delete') {
   223→      setEstimates((prev) => prev.filter((e) => !selectedIds.includes(e.id)));
   224→      setSelectedIds([]);
   225→      onToast?.(`Deleted ${count} estimate(s)`);
   226→      return;
   227→    }
   228→    if (action === 'convert') {
   229→      setEstimates((prev) =>
   230→        prev.map((e) =>
   231→          selectedIds.includes(e.id)
   232→            ? { ...e, status: 'Converted', convertedInvoiceNo: `10${Math.floor(Math.random() * 90) + 10}` }
   233→            : e
   234→        )
   235→      );
   236→      setSelectedIds([]);
   237→      onToast?.(`Converted ${count} estimate(s) to invoice`);
   238→      return;
   239→    }
   240→    setSelectedIds([]);
   241→    onToast?.(`${action.charAt(0).toUpperCase() + action.slice(1)} on ${count} estimate(s)`);
   242→  };

   244→  // Open Drawer for Create / Edit
   245→  const handleOpenDrawer = (est: EstimateItem | null) => {
   246→    if (est) {
   247→      setEditingEst({ ...est });
   248→    } else {
   249→      const nextNo = `02${estimates.length + 20}`;
   250→      setEditingEst({
   251→        id: `est-${Date.now()}`,
   252→        no: nextNo,
   253→        customer: 'Acme Corp',
   254→        email: 'ap@acme.com',
   255→        avatarLetter: 'A',
   256→        avatarClass: styles.avatarBlue,
   257→        sentDate: '11/04/2026',
   258→        expiryDate: '12/04/2026',
   259→        amount: 2800,
   260→        status: 'Draft',
   261→        note: 'Thanks for the opportunity — let us know if you have questions.',
   262→        lines: [
   263→          { id: 'l1', desc: 'Custom design — 3 concepts', qty: 3, rate: 800.0 },
   264→          { id: 'l2', desc: 'Print setup + samples', qty: 1, rate: 400.0 },
   265→        ],
   266→      });
   267→    }
   268→    setDrawerOpen(true);
   269→  };
   270→
   271→  const handleDrawerSave = () => {
   272→    if (!editingEst) return;
   273→    const lineTotal = editingEst.lines.reduce((acc, l) => acc + l.qty * l.rate, 0);
   274→    const updated = { ...editingEst, amount: lineTotal };
   275→
   276→    setEstimates((prev) => {
   277→      const exists = prev.some((e) => e.id === updated.id);
   278→      if (exists) {
   279→        return prev.map((e) => (e.id === updated.id ? updated : e));
   280→      }
   281→      return [updated, ...prev];
   282→    });
   283→    setDrawerOpen(false);
   284→    onToast?.(`Estimate ${updated.no} saved`);
   285→  };
   286→
   287→  const handleDrawerSend = () => {
   288→    if (!editingEst) return;
   289→    const lineTotal = editingEst.lines.reduce((acc, l) => acc + l.qty * l.rate, 0);
   290→    const updated: EstimateItem = { ...editingEst, amount: lineTotal, status: 'Sent' };
   291→
   292→    setEstimates((prev) => {
   293→      const exists = prev.some((e) => e.id === updated.id);
   294→      if (exists) {
   295→        return prev.map((e) => (e.id === updated.id ? updated : e));
   296→      }
   297→      return [updated, ...prev];
   298→    });
   299→    setDrawerOpen(false);
   300→    onToast?.(`Estimate ${updated.no} sent to customer`);
   301→  };
   302→
   303→  const handlePromptConvert = (est: EstimateItem) => {
   304→    setConvertTargetEst(est);
   305→    setConvertModalOpen(true);
   306→  };
   307→
   308→  const handleConfirmConvert = () => {
   309→    if (!convertTargetEst) return;
   310→    const newInvoiceNo = `10${Math.floor(Math.random() * 90) + 10}`;
   311→    setEstimates((prev) =>
   312→      prev.map((e) =>
   313→        e.id === convertTargetEst.id
   314→          ? { ...e, status: 'Converted', convertedInvoiceNo: newInvoiceNo }
   315→          : e
   316→      )
   317→    );
   318→    setConvertModalOpen(false);
   319→    setDrawerOpen(false);
   320→    onToast?.(`Estimate ${convertTargetEst.no} converted to invoice ${newInvoiceNo}`);
   321→  };
   322→
   323→  // Row More Menu
   324→  const handleOpenRowMenu = (e: React.MouseEvent, est: EstimateItem) => {
   325→    e.stopPropagation();
   326→    const rect = e.currentTarget.getBoundingClientRect();
   327→    setMenuPos({
   328→      top: rect.bottom + 4,
   329→      left: Math.min(rect.right - 230, window.innerWidth - 250),
   330→    });
   331→    setActiveMenuEst(est);
   332→    setMenuOpen(true);
   333→  };
   334→
   335→  const handleRowMenuAction = (action: string) => {
   336→    if (!activeMenuEst) return;
   337→    const est = activeMenuEst;
   338→    setMenuOpen(false);
   339→
   340→    switch (action) {
   341→      case 'view':
   342→      case 'edit':
   343→        handleOpenDrawer(est);
   344→        break;
   345→      case 'duplicate': {
   346→        const copy: EstimateItem = {
   347→          ...est,
   348→          id: `est-${Date.now()}`,
   349→          no: `${est.no}-copy`,
   350→          status: 'Draft',
   351→        };
   352→        setEstimates((prev) => [copy, ...prev]);
   353→        onToast?.(`Duplicated estimate ${est.no}`);
   354→        break;
   355→      }
   356→      case 'send':
   357→        setEstimates((prev) =>
   358→          prev.map((e) => (e.id === est.id ? { ...e, status: 'Sent' } : e))
   359→        );
   360→        onToast?.(`Sent estimate ${est.no}`);
   361→        break;
   362→      case 'remind':
   363→        onToast?.(`Reminder sent for estimate ${est.no}`);
   364→        break;
   365→      case 'print':
   366→        onToast?.(`Printing estimate ${est.no} PDF`);
   367→        break;
   368→      case 'convert':
   369→        handlePromptConvert(est);
   370→        break;
   371→      case 'mark-accepted':
   372→        setEstimates((prev) =>
   373→          prev.map((e) => (e.id === est.id ? { ...e, status: 'Accepted' } : e))
   374→        );
   375→        onToast?.(`Estimate ${est.no} marked accepted`);
   376→        break;
   377→      case 'mark-declined':
   378→        setEstimates((prev) =>
   379→          prev.map((e) => (e.id === est.id ? { ...e, status: 'Declined' } : e))
   380→        );
   381→        onToast?.(`Estimate ${est.no} marked declined`);
   382→        break;
   383→      case 'delete':
   384→        setEstimates((prev) => prev.filter((e) => e.id !== est.id));
   385→        onToast?.(`Deleted estimate ${est.no}`);
   386→        break;
   387→    }
   388→  };

   392→      style: 'currency',
   393→      currency: 'USD',
   394→      minimumFractionDigits: 2,
   395→    }).format(amt);
   396→  };

   401→        <span className={`${styles.statusPill} ${styles.statusConverted}`}>

   407→      case 'Draft':
   408→        return <span className={`${styles.statusPill} ${styles.statusDraft}`}>Draft</span>;

   410→        return <span className={`${styles.statusPill} ${styles.statusSent}`}>Sent</span>;

   412→        return <span className={`${styles.statusPill} ${styles.statusViewed}`}>Viewed</span>;

   414→        return <span className={`${styles.statusPill} ${styles.statusAccepted}`}>Accepted</span>;
   415→      case 'Declined':
   416→        return <span className={`${styles.statusPill} ${styles.statusDeclined}`}>Declined</span>;
   417→      case 'Expired':
   418→        return <span className={`${styles.statusPill} ${styles.statusExpired}`}>Expired</span>;
   419→      default:
   420→        return <span className={`${styles.statusPill} ${styles.statusDraft}`}>{est.status}</span>;
   421→    }
   422→  };

   425→    <div className={styles.container}>
   426→      {/* Context Header */}
   427→      <div className={styles.contextHeader}>
   428→        <div className={styles.breadcrumbGroup}>
   429→          <span className={`material-symbols-outlined ${styles.domainIcon}`}>
   430→            account_balance
   431→          </span>
   432→          <span className={styles.breadcrumbLabel}>Finance</span>
   433→          <span className={styles.breadcrumbSep}>/</span>
   434→          <span className={styles.breadcrumbLabel}>Sales &amp; Get Paid</span>
   435→          <span className={styles.breadcrumbSep}>/</span>
   436→          <span className={styles.breadcrumbActive}>Estimates</span>
   437→        </div>
   438→        <div className={styles.previewBadge}>Quotes · standalone</div>
   439→      </div>
   440→

   444→          <div>
   445→            <h1 className={styles.title}>Estimates</h1>
   446→            <p className={styles.subtitle}>
   447→              Quotes sent to customers · convert to invoice when accepted
   448→            </p>
   449→          </div>
   450→
   451→          <div className={styles.headerActions}>
   452→            <div className={styles.splitBtn}>
   453→              <button
   454→                type="button"
   455→                className={styles.splitBtnMain}
   456→                onClick={() => handleOpenDrawer(null)}
   457→              >
   458→                <span className="material-symbols-outlined" style={{ fontSize: 16 }}>
   459→                  add
   460→                </span>
   461→                New estimate
   462→              </button>
   463→              <button
   464→                ref={newSplitBtnRef}
   465→                type="button"
   466→                className={styles.splitBtnArrow}
   467→                onClick={(e) => {
   468→                  e.stopPropagation();
   469→                  const rect = e.currentTarget.getBoundingClientRect();
   470→                  setNewMenuPos({
   471→                    top: rect.bottom + 4,
   472→                    left: Math.min(rect.right - 220, window.innerWidth - 240),
   473→                  });
   474→                  setNewMenuOpen((prev) => !prev);
   475→                }}
   476→              >
   477→                <span className="material-symbols-outlined" style={{ fontSize: 16 }}>
   478→                  expand_more
   479→                </span>
   480→              </button>
   481→            </div>
   482→
   483→            <button
   484→              type="button"
   485→              className={styles.secondaryBtn}
   486→              onClick={() => onToast?.('Exporting estimates…')}
   487→            >
   488→              <span className="material-symbols-outlined" style={{ fontSize: 16 }}>
   489→                download
   490→              </span>
   491→              Export
   492→            </button>
   493→
   494→            <button
   495→              type="button"
   496→              className={styles.secondaryBtn}
   497→              onClick={() => onToast?.('Customize panel opened')}
   498→            >
   499→              <span className="material-symbols-outlined" style={{ fontSize: 16 }}>
