    35→function buildWorkspaceTab(domain: DomainKey, screenId: string): WorkspaceTab {
    36→  const scr = findScreen(domain, screenId);
    37→  const label = scr.label === 'Overview' ? `${scr.group} · Overview` : scr.label;
    38→  return {
    39→    key: `${domain}:${scr.id}`,
    40→    domain,
    41→    screenId: scr.id,
    42→    label,
    43→  };
    44→}

    80→function loadPinMode(): PinMode {
    81→  if (typeof window !== 'undefined') {
    82→    try {
    83→      const saved = localStorage.getItem(PIN_MODE_STORAGE_KEY);
    84→      if (saved === 'bookmarks' || saved === 'favorites') return saved;
    85→    } catch {
    86→      // ignore
    87→    }
    88→  }
    89→  return 'bookmarks';
    90→}
    91→
    92→function loadBookmarks(): BookmarkEntry[] {
    93→  if (typeof window !== 'undefined') {
    94→    try {
    95→      const saved = localStorage.getItem(BOOKMARKS_STORAGE_KEY);
    96→      if (saved) {
    97→        const parsed = JSON.parse(saved);
    98→        if (Array.isArray(parsed)) {
    99→          return parsed.filter(
   100→            (b): b is BookmarkEntry =>
   101→              b &&
   102→              typeof b.domain === 'string' &&
   103→              typeof b.screen === 'string' &&
   104→              !!DOMAINS[b.domain as DomainKey]
   105→          );
   106→        }
   107→      }
   108→    } catch {
   109→      // ignore
   110→    }
   111→  }
   112→  return DEFAULT_BOOKMARKS;
   113→}

   595→
   596→  const handleRailAction = useCallback(
   597→    (id: string) => {
   598→      if (id === 'allapps') {
   599→        // All apps → Enterprise Navigation Hub (routing), NOT create.
   600→        setNavigationHubOpen(true);
   601→      } else if (id === 'customize' || id === 'favorites') {
   602→        // The star (favorites/bookmarks) pin toggles the favorites flyout.
   603→        setFavoritesFlyoutOpen((prev) => !prev);
   604→      } else if (id === 'reports') {
   605→        // Reports is its own standalone, cross-module view — NOT tied to
   606→        // accounting/finance. Broken down by every module inside the reports
   607→        // domain.
   608→        openOrActivateTab('reports', 'reports-overview');
   609→      } else if (id === 'inbox') {
   610→        // Inbox → Support omnichannel inbox (real screen).
   611→        openOrActivateTab('support', 'messages');
   612→      } else if (id === 'tasks') {
   613→        showToast('Tasks & calendar');
   614→      } else if (id === 'home') {
   615→        openOrActivateTab('dashboard', 'ops');
   616→      } else {
   617→        showToast(`Rail → ${id}`);
   618→      }
   619→    },
   620→    [openOrActivateTab, showToast]
   621→  );
   622→
   623→  const handleUtilAction = useCallback(
   624→    (id: string) => {
   625→      if (id === 'settings') {
   626→        openOrActivateTab('settings', 'ws-home');
   627→        setCreateFlyoutOpen(false);
   628→      } else {
   629→        showToast(`Utility → ${id}`);
   630→      }
   631→    },
   632→    [openOrActivateTab, showToast]
   633→  );
   634→
   635→  // ── Page-level view controls (below the tab bar, on every page) ──
   636→  const handleOpenStandalone = useCallback(() => {
   637→    window.open(`/?view=${activeDomain}:${activeScreen}&focus=1`, '_blank');
   638→  }, [activeDomain, activeScreen]);
   639→
   640→  const handleToggleFocus = useCallback(() => {
   641→    setFocusMode((prev) => {
   642→      const next = !prev;
   643→      showToast(next ? 'Full screen — press Esc to exit' : 'Exited full screen');
   644→      return next;
   645→    });
   646→  }, [showToast]);
   647→
   648→  // ── Dual pin handlers ──
   649→  const handlePinModeChange = useCallback((mode: PinMode) => {
   650→    setPinMode(mode);
   651→    try {
   652→      localStorage.setItem(PIN_MODE_STORAGE_KEY, mode);
   653→    } catch {
   654→      // ignore
   655→    }
   656→  }, []);
   657→
   658→  const handleAddBookmark = useCallback((domain: DomainKey, screen: string) => {
   659→    setRailBookmarks((prev) => {
   660→      if (prev.some((b) => b.domain === domain && b.screen === screen)) return prev;
   661→      const next = [...prev, { domain, screen }];
   662→      try {
   663→        localStorage.setItem(BOOKMARKS_STORAGE_KEY, JSON.stringify(next));
   664→      } catch {
   665→        // ignore
   666→      }
   667→      return next;
   668→    });
   669→  }, []);
   670→
   671→  const handleRemoveBookmark = useCallback((domain: DomainKey, screen: string) => {
   672→    setRailBookmarks((prev) => {
   673→      const next = prev.filter((b) => !(b.domain === domain && b.screen === screen));
   674→      try {
   675→        localStorage.setItem(BOOKMARKS_STORAGE_KEY, JSON.stringify(next));
   676→      } catch {
   677→        // ignore
   678→      }
   679→      return next;
   680→    });
   681→  }, []);
   682→
   683→  const activeTabKey = `${activeDomain}:${activeScreen}`;
   684→
   685→  return (
   686→    <>
   687→      <Header
   688→        openTabs={openTabs}
   689→        activeTabKey={activeTabKey}
   690→        onSelectTab={handleSelectTab}
   691→        onCloseTab={handleCloseTab}
   692→        onOpenCreateFlyout={() => setCreateFlyoutOpen(true)}
   693→        onOpenNavigationHub={() => setNavigationHubOpen(true)}
   694→        onOpenSettings={() => openOrActivateTab('settings', 'ws-home')}
   695→        viewMode={viewMode}
   696→        onChangeViewMode={(mode) => {
   697→          setViewMode(mode);
   698→          showToast(`Switched to ${mode} view`);
   699→        }}
   700→        onToast={showToast}
   701→        hideTabs={focusMode}
   702→      />
   703→
   704→      {!focusMode && (
   705→        <MainRail
   706→          activeDomain={activeDomain}
   707→          onSelectDomain={handleSelectDomain}
   708→          onOpenCreateFlyout={() => setCreateFlyoutOpen(true)}
   709→          onRailAction={handleRailAction}
   710→          onUtilAction={handleUtilAction}
   711→          favoritesCount={favoriteIds.length}
   712→          pinMode={pinMode}
   713→          bookmarksCount={railBookmarks.length}
   714→          favoritesOpen={favoritesFlyoutOpen}
   715→          onToggleFavorites={() => setFavoritesFlyoutOpen((prev) => !prev)}
   716→        />
   717→      )}
   718→
   719→      {!focusMode && activeDomain !== 'dashboard' && (
   720→        <SideRail
   721→          activeDomain={activeDomain}
   722→          activeScreen={activeScreen}
   723→          openGroups={openGroupsByDomain[activeDomain] || []}
   724→          sideFilter={sideFilter}
   725→          onSideFilterChange={setSideFilter}
   726→          onToggleGroup={handleToggleGroup}
   727→          onSelectScreen={handleSelectScreen}
   728→          collapsed={sideRailCollapsed}
   729→          onToggleCollapse={() => setSideRailCollapsed((prev) => !prev)}
   730→          screenBadges={dashboardController.queueBadges}
   731→        />
   732→      )}
   733→
   734→      <ScreenContent
   735→        openTabs={openTabs}
   736→        activeTabKey={activeTabKey}
   737→        dashboardController={dashboardController}
   738→        onNavigate={openOrActivateTab}
   739→        onCapCreate={handleCapCreate}
   740→        onCreate={handleCreate}
   741→        onOpenCreateFlyout={() => setCreateFlyoutOpen(true)}
   742→        onOpenPeek={(row) => {
   743→          setPeekRow(row);
   744→          setPeekOpen(true);
   745→        }}
   746→        onToast={showToast}
   747→        sideRailCollapsed={activeDomain === 'dashboard' ? true : sideRailCollapsed}
   748→        focusMode={focusMode}
   749→        onToggleFocus={handleToggleFocus}
   750→        onOpenStandalone={handleOpenStandalone}
   751→      />
   752→
   753→      <DashboardOverlays controller={dashboardController} onToast={showToast} />
   754→
   755→      <PeekDrawer
   756→        open={peekOpen}
   757→        row={peekRow}
   758→        onClose={() => setPeekOpen(false)}
   759→        onOpenFull={() => {
   760→          setPeekOpen(false);
   761→          openDocEngine();
   762→        }}
   763→      />
   764→
   765→      <RecordDrawer
   766→        open={recordDrawerOpen}
   767→        kind={recordKind}
   768→        sessionKey={recordSessionKey}
   769→        onClose={() => setRecordDrawerOpen(false)}
   770→      />
   771→
   772→      <DocumentEngine
   773→        open={docEngineOpen}
   774→        activeTab={docTab}
   775→        onSelectTab={setDocTab}
   776→        onClose={() => setDocEngineOpen(false)}
   777→        onToast={showToast}
   778→      />
   779→
   780→      <CreateFlyout
   781→        open={createFlyoutOpen}
   782→        onClose={() => setCreateFlyoutOpen(false)}
   783→        onSelectCreate={handleCreate}
   784→      />
   785→
   786→      <NavigationHubFlyout
   787→        open={navigationHubOpen}
   788→        onClose={() => setNavigationHubOpen(false)}
   789→        onNavigate={openOrActivateTab}
   790→        onToast={showToast}
   791→      />
   792→
   793→      <StudioDrawer
   794→        open={studioDrawerOpen}
   795→        kind={studioDrawerKind}
   796→        onClose={() => setStudioDrawerOpen(false)}
   797→        onToast={showToast}
   798→      />
   799→      <QuickActionsDrawer
   800→        open={quickActionsDrawerOpen}
   801→        onClose={() => setQuickActionsDrawerOpen(false)}
   802→        selectedActionIds={favoriteIds}
   803→        onSelectedActionIdsChange={(newIds) => {
   804→          setFavoriteIds(newIds);
   805→          try {
   806→            localStorage.setItem(FAVORITES_STORAGE_KEY, JSON.stringify(newIds));
   807→          } catch {
   808→            // ignore
   809→          }
   810→        }}
   811→        onToast={showToast}
   812→        onSelectAction={(actionName) => {
   813→          handleCreate(actionName);
   814→        }}
   815→      />
   816→
   817→      <FavoritesFlyout
   818→        open={favoritesFlyoutOpen}
   819→        pinMode={pinMode}
   820→        onPinModeChange={handlePinModeChange}
   821→        bookmarks={railBookmarks}
   822→        activeDomain={activeDomain}
   823→        activeScreen={activeScreen}
   824→        onAddBookmark={handleAddBookmark}
   825→        onRemoveBookmark={handleRemoveBookmark}
   826→        onNavigateBookmark={openOrActivateTab}
   827→        favoriteIds={favoriteIds}
   828→        onClose={() => setFavoritesFlyoutOpen(false)}
   829→        onSelectAction={(actionName) => {
   830→          handleCreate(actionName);
   831→        }}
   832→        onOpenCustomize={() => {
   833→          setFavoritesFlyoutOpen(false);
   834→          setQuickActionsDrawerOpen(true);
   835→        }}
   836→        onToast={showToast}
   837→      />
   838→
   839→      <Toast message={toastMsg} visible={toastVisible} />
   840→    </>
   841→  );
   842→}
