   733→export function screensOf(domain: DomainConfig): ScreenWithGroup[] {
   734→  return domain.groups.flatMap((g) => g.screens.map((s) => ({ ...s, group: g.name })));
   735→}
   736→
   737→export function findScreen(domainId: DomainKey, screenId: string): ScreenWithGroup {
   738→  const all = screensOf(DOMAINS[domainId]);
   739→  return all.find((s) => s.id === screenId) || all[0];
   740→}
