   667→      html += `<tr class="sub">${row.cells.map((c,i)=>`<td class="${i===0?"":"right"}">${formatNeg(c)}</td>`).join("")}</tr>`;
   668→    } else if (row.kind === "total"){
   669→      const label = row.label || "";
   670→      const cells = row.cells.map((c,i)=>`<td class="right">${formatNeg(c)}</td>`).join("");
   671→      html += `<tr class="total"><td>${label}</td>${cells}</tr>`;

   744→let dragSrc = null;
   745→document.querySelectorAll(".col-item").forEach(item=>{
   746→  item.addEventListener("dragstart", e=>{ dragSrc = item; item.classList.add("dragging"); });
   747→  item.addEventListener("dragend", ()=>{ item.classList.remove("dragging"); dragSrc = null; });
   748→  item.addEventListener("dragover", e=>{
   749→    e.preventDefault();
   750→    if (!dragSrc || dragSrc === item) return;
   751→    const rect = item.getBoundingClientRect();
   752→    const after = (e.clientY - rect.top) > rect.height/2;
   753→    item.parentNode.insertBefore(dragSrc, after ? item.nextSibling : item);
   754→  });
   755→});
